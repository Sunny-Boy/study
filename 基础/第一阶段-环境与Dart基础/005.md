### 1. 异步编程基础概念

#### 1.1 同步vs异步编程

```dart
// 同步编程示例（阻塞式）
void synchronousExample() {
  print('开始执行同步操作');
  
  // 模拟耗时操作（会阻塞线程）
  for (int i = 0; i < 1000000000; i++) {
    // 大量计算，阻塞主线程
  }
  
  print('同步操作完成');
  print('后续代码执行'); // 必须等待上面的操作完成
}

// 异步编程示例（非阻塞式）
void asynchronousExample() {
  print('开始执行异步操作');
  
  // 异步操作不会阻塞主线程
  Future.delayed(Duration(seconds: 2), () {
    print('异步操作完成');
  });
  
  print('后续代码立即执行'); // 立即执行，不等待异步操作
}

void compareApproaches() {
  print('=== 同步编程 ===');
  synchronousExample();
  
  print('\n=== 异步编程 ===');  
  asynchronousExample();
  
  // 输出顺序演示异步的特性
}
```

```dart
class FutureBasics {

  static Future<String> getImmediateResult() {
    return Future.value('立即返回的结果');
  }
  
  // 2. 使用Future.delayed创建延迟Future
  static Future<String> getDelayedResult() {
    return Future.delayed(

  // 3. 使用Completer手动控制Future
  static Future<int> getControlledResult() {
    Completer<int> completer = Completer<int>();
    
    // 模拟异步操作
    Timer(Duration(seconds: 1), () {
      Random random = Random();
      if (random.nextBool()) {
        completer.complete(42); // 成功完成
      } else {
        completer.completeError('操作失败'); // 完成并带错误
      }
    });
    
    return completer.future;
  }
  
  // 4. 使用Future构造函数
  static Future<List<String>> getComputedResult() {
    return Future(() {
      // 模拟计算密集型任务
      List<String> results = [];
      for (int i = 0; i < 1000; i++) {
        results.add('Item $i');
      }
      return results;
    });
  }
}
```

#### 2.2 Future错误处理

```dart
class FutureErrorHandling {
  // 可能失败的异步操作
  static Future<String> riskyOperation(bool shouldFail) {
    return Future.delayed(Duration(seconds: 1), () {
      if (shouldFail) {
        throw Exception('操作失败了！');
      }
      return '操作成功完成';
    });
  }
  
  // 方法1：使用then/catchError处理错误
  static void handleWithThenCatch() {
    print('📛 使用then/catchError处理错误:');
    
    riskyOperation(true)
      .then((result) {
        print('成功: $result');
      })
      .catchError((error) {
        print('错误: $error');
        return '默认值'; // 可以返回默认值
      })
      .then((finalResult) {
        print('最终结果: $finalResult');
      });
  }
  
  // 方法2：使用Future.onError处理错误
  static void handleWithOnError() {
    print('🛠️ 使用onError处理错误:');
    
    riskyOperation(true)
      .onError<Exception>((error, stackTrace) {
        print('捕获到异常: $error');
        print('堆栈跟踪: $stackTrace');
        return '错误恢复值';
      })
      .then((result) {
        print('处理后结果: $result');
      });
  }
  
  // 方法3：使用try-catch (需要在async函数中)
  static Future<void> handleWithTryCatch() async {
    print('⚡ 使用try-catch处理错误:');
    
    try {
      String result = await riskyOperation(true);
      print('成功: $result');
    } on Exception catch (e) {
      print('捕获到特定异常: $e');
    } catch (e, stackTrace) {
      print('捕获到通用错误: $e');
      print('堆栈跟踪: $stackTrace');
    } finally {
      print('清理代码（总是执行）');
    }
  }
}
```

```dart
class AsyncFlowControl {
  // 条件异步执行
  static Future<String> conditionalAsync(bool useCache) async {
    print('🔍 开始条件异步执行...');
    
    if (useCache) {
      print('📦 使用缓存数据');
      return await getCachedData();
    } else {
      print('🌐 从网络获取数据');
      String networkData = await fetchFromNetwork();
      await cacheData(networkData);
      return networkData;
    }
  }
  
  // 循环中的异步操作
  static Future<List<String>> processItemsSequentially(List<String> items) async {
    print('📝 顺序处理${items.length}个项目:');
    List<String> results = [];
    
    for (String item in items) {
      print('处理项目: $item');
      String processed = await processItem(item);
      results.add(processed);
    }
    
    return results;
  }
  
  // 并发处理多个项目
  static Future<List<String>> processItemsConcurrently(List<String> items) async {
    print('⚡ 并发处理${items.length}个项目:');
    
    List<Future<String>> futures = items.map((item) => processItem(item)).toList();
    return await Future.wait(futures);
  }
  
  // 带重试机制的异步操作
  static Future<String> withRetry(int maxRetries) async {
    int attempts = 0;
    
    while (attempts < maxRetries) {
      attempts++;
      print('第$attempts次尝试...');
      
      try {
        return await unreliableOperation();
      } catch (error) {
        print('尝试失败: $error');
        
        if (attempts >= maxRetries) {
          throw Exception('达到最大重试次数，操作失败');
        }
        
        // 等待后重试
        await Future.delayed(Duration(seconds: attempts));
      }
    }
    
    throw Exception('意外错误'); // 不应该到这里
  }
  
  // 辅助方法
  static Future<String> getCachedData() => 
    Future.delayed(Duration(milliseconds: 100), () => 'cached_data');
  
  static Future<String> fetchFromNetwork() => 
    Future.delayed(Duration(seconds: 2), () => 'network_data');
  
  static Future<void> cacheData(String data) => 
    Future.delayed(Duration(milliseconds: 500), () => print('数据已缓存'));
  
  static Future<String> processItem(String item) => 
    Future.delayed(Duration(milliseconds: 500), () => 'processed_$item');
  
  static Future<String> unreliableOperation() => 
    Future.delayed(Duration(seconds: 1), () {
      if (Random().nextBool()) {
        throw Exception('随机失败');
      }
      return '操作成功';
    });
}
```

```dart
class AsyncReturnTypes {
  // 返回Future<void>的异步函数
  static Future<void> performTask() async {
    print('开始执行任务...');
    await Future.delayed(Duration(seconds: 1));
    print('任务完成');
    // 无需显式返回，自动返回Future<void>
  }
  
  // 返回Future<T>的异步函数
  static Future<int> calculateSum(List<int> numbers) async {
    print('开始计算总和...');
    await Future.delayed(Duration(milliseconds: 500));
    
    int sum = numbers.fold(0, (prev, curr) => prev + curr);
    return sum; // 自动包装为Future<int>
  }
  
  // 返回FutureOr<T>的函数（可能同步也可能异步）
  static FutureOr<String> getResult(bool useAsync) {
    if (useAsync) {
      return Future.delayed(Duration(seconds: 1), () => '异步结果');
    } else {
      return '同步结果'; // 直接返回，不包装为Future
    }
  }
  
  // 异步生成器函数
  static Stream<int> countNumbers(int max) async* {
    for (int i = 1; i <= max; i++) {
      await Future.delayed(Duration(milliseconds: 100));
      yield i; // 生成下一个值
    }
  }
  
  // 使用示例
  static Future<void> demonstrateReturnTypes() async {
    // 1. void异步函数
    await performTask();
    
    // 2. 有返回值的异步函数
    int sum = await calculateSum([1, 2, 3, 4, 5]);
    print('计算结果: $sum');
    
    // 3. FutureOr函数
    String syncResult = await getResult(false);
    print('同步结果: $syncResult');
    
    String asyncResult = await getResult(true);
    print('异步结果: $asyncResult');
    
    // 4. 异步流
    print('数字流:');
    await for (int number in countNumbers(5)) {
```

```dart
    _controller.close();
  }
  
  // 4. 使用async*创建异步生成器
```

```dart
    }
  }

// 自定义Stream扩展方法
extension StreamExtensions<T> on Stream<T> {
  // 缓冲指定数量的元素
  Stream<List<T>> bufferCount(int count) async* {
    List<T> buffer = [];
    await for (T item in this) {
      buffer.add(item);
      if (buffer.length == count) {
        yield List.from(buffer);
        buffer.clear();
      }
    }
    if (buffer.isNotEmpty) {
      yield buffer;
    }
  }
  
  // 按时间缓冲元素
  Stream<List<T>> bufferTime(Duration duration) async* {
    List<T> buffer = [];
    Timer? timer;
    StreamSubscription<T>? subscription;
    late StreamController<List<T>> controller;
    
    controller = StreamController<List<T>>(
      onListen: () {
        timer = Timer.periodic(duration, (t) {
          if (buffer.isNotEmpty) {
            controller.add(List.from(buffer));
            buffer.clear();
          }
        });
        
        subscription = this.listen(
          (item) => buffer.add(item),
          onError: controller.addError,
          onDone: () {
            timer?.cancel();
            if (buffer.isNotEmpty) {
              controller.add(buffer);
            }
            controller.close();
          },
        );
      },
      onCancel: () {
        timer?.cancel();
        subscription?.cancel();
      },
    );
    
    yield* controller.stream;
  }
  
  // 错误时返回默认值
  Stream<T> onErrorReturn(T defaultValue) async* {
    await for (T item in this) {
      try {
        yield item;
      } catch (e) {
        yield defaultValue;
      }
    }
  }
}
```

## 大白话解释

Dart异步编程就像是在经营一家**餐厅**：

- **同步编程** = **单线厨师**：一次只能做一道菜，其他订单要排队等待
- **异步编程** = **智能厨房**：同时准备多道菜，合理安排时间

**Future** 就像是**订单小票**：
- 顾客下单后得到小票（Future）
- 小票显示"正在制作中"（pending状态）
- 菜做好了小票变成"已完成"（completed状态）
- 如果出错小票变成"制作失败"（error状态）

**Stream** 就像是**传送带**：
- 厨房不断地把做好的菜放到传送带上
- 服务员在传送带旁等着取菜（监听）
- 可以暂停传送带（pause）或让它继续转（resume）

**Event Loop** 就像是**餐厅经理**：
- 优先处理紧急事务（微任务队列）
- 然后按顺序处理普通事务（事件队列）
- 确保餐厅运转顺畅

```dart
    return ApiResponse(

    );
  }
  
  factory ApiResponse.error(String error, int statusCode) {

}

class NetworkManager {

    _httpClient = HttpClient()
      ..connectionTimeout = defaultTimeout
      ..idleTimeout = defaultTimeout;
  }
  
  // 通用GET请求
  Future<ApiResponse<Map<String, dynamic>>> get(
    String endpoint, {
    Map<String, String>? headers,
    Map<String, String>? queryParams,
  }) async {
    try {
      String url = _buildUrl(endpoint, queryParams);

      // 设置请求头
      _setHeaders(request, headers);
      
      // 发送请求并获取响应
      HttpClientResponse response = await request.close();
      
      return await _handleResponse(response);
    } catch (error) {
      print('❌ GET请求失败: $error');
      return ApiResponse.error('网络连接失败: $error', -1);
    }
  }
  
  // 通用POST请求
  Future<ApiResponse<Map<String, dynamic>>> post(
    String endpoint, {
    Map<String, dynamic>? body,

  }) async {
    try {
      String url = _buildUrl(endpoint);

      if (body != null) {

      }
      
      // 发送请求并获取响应
      HttpClientResponse response = await request.close();
      
      return await _handleResponse(response);
    } catch (error) {
      print('❌ POST请求失败: $error');
      return ApiResponse.error('网络连接失败: $error', -1);
    }
  }
  
  // 带重试机制的请求
  Future<ApiResponse<Map<String, dynamic>>> requestWithRetry(
    Future<ApiResponse<Map<String, dynamic>>> Function() requestFunc, {
    int maxRetries = 3,
    Duration retryDelay = const Duration(seconds: 1),
  }) async {
    int attempts = 0;
    
    while (attempts < maxRetries) {
      attempts++;
      print('🔄 第$attempts次尝试...');
      
      ApiResponse<Map<String, dynamic>> response = await requestFunc();
      
      if (response.success || attempts >= maxRetries) {
        return response;
      }
      
      print('⚠️ 请求失败，${retryDelay.inSeconds}秒后重试...');
      await Future.delayed(retryDelay * attempts); // 指数退避
    }
    
    return ApiResponse.error('达到最大重试次数', -1);
  }
  
  // 并发请求
  Future<List<ApiResponse<Map<String, dynamic>>>> concurrentRequests(
    List<String> endpoints,
  ) async {
    print('⚡ 并发请求${endpoints.length}个接口...');
    
    List<Future<ApiResponse<Map<String, dynamic>>>> futures = 
      endpoints.map((endpoint) => get(endpoint)).toList();
    
    return await Future.wait(futures);
  }
  
  // 请求取消支持
  Future<ApiResponse<Map<String, dynamic>>> cancellableRequest(

    required Duration timeout,
  }) async {
    try {
      return await get(endpoint).timeout(
        timeout,
        onTimeout: () {
          print('⏰ 请求超时: $endpoint');
          return ApiResponse.error('请求超时', 408);
        },
      );
    } catch (error) {
      return ApiResponse.error('请求被取消: $error', -1);
    }
  }
  
  // 私有辅助方法
  String _buildUrl(String endpoint, [Map<String, String>? queryParams]) {
    String url = '$baseUrl$endpoint';
    
    if (queryParams != null && queryParams.isNotEmpty) {
      String query = queryParams.entries
        .map((entry) => '${entry.key}=${Uri.encodeComponent(entry.value)}')
        .join('&');
      url += '?$query';
    }
    
    return url;
  }
  
  void _setHeaders(HttpClientRequest request, Map<String, String>? headers) {
    // 默认请求头
    request.headers.set('User-Agent', 'Dart-App/1.0');
    request.headers.set('Accept', 'application/json');
    
    // 自定义请求头
    headers?.forEach((key, value) {
      request.headers.set(key, value);
    });
  }

  ) async {
    String responseBody = await response.transform(utf8.decoder).join();
    
    print('📨 响应状态: ${response.statusCode}');
    print('📄 响应内容: $responseBody');
    
    if (response.statusCode >= 200 && response.statusCode < 300) {
      try {
        Map<String, dynamic> data = json.decode(responseBody);
        return ApiResponse.success(data, response.statusCode);
      } catch (error) {
        return ApiResponse.error('响应解析失败: $error', response.statusCode);
      }
    } else {
      return ApiResponse.error(
        'HTTP错误 ${response.statusCode}: $responseBody',
        response.statusCode,
      );
    }
  }
  
  void dispose() {
    _httpClient.close();
  }
}

class CacheManager<T> {
  final Map<String, CacheItem<T>> _cache = {};
  final Duration _defaultTtl = Duration(minutes: 5);
  
  // 缓存项
  class CacheItem<T> {
    final T data;
    final DateTime expiry;
    
    CacheItem(this.data, this.expiry);
    
    bool get isExpired => DateTime.now().isAfter(expiry);
  }
  
  // 获取缓存数据
  Future<T?> get(String key) async {
    CacheItem<T>? item = _cache[key];
    
    if (item == null || item.isExpired) {
      _cache.remove(key);
      return null;
    }
    
    return item.data;
  }
  
  // 设置缓存数据
  Future<void> set(String key, T data, [Duration? ttl]) async {
    Duration actualTtl = ttl ?? _defaultTtl;
    DateTime expiry = DateTime.now().add(actualTtl);
    
    _cache[key] = CacheItem<T>(data, expiry);
  }
  
  // 清理过期缓存
  void cleanExpired() {
    _cache.removeWhere((key, item) => item.isExpired);
  }
  
  // 清空所有缓存
  void clear() {
    _cache.clear();
  }
}
```

```dart
// 实时数据分析器
class RealTimeAnalyzer {
  final List<SensorData> _dataBuffer = [];
  final int _bufferSize;
  final StreamController<AnalysisResult> _resultController;
  
  RealTimeAnalyzer({int bufferSize = 10})
    : _bufferSize = bufferSize,
      _resultController = StreamController<AnalysisResult>.broadcast();
  
  // 分析结果流
  Stream<AnalysisResult> get analysisResults => _resultController.stream;
  
  // 处理传感器数据
  void processSensorData(SensorData data) {
    // 添加到缓冲区
    _dataBuffer.add(data);
    
    // 保持缓冲区大小
    if (_dataBuffer.length > _bufferSize) {
      _dataBuffer.removeAt(0);
    }
    
    // 执行分析
    _performAnalysis(data.type);
  }
  
  void _performAnalysis(String dataType) {
    if (_dataBuffer.isEmpty) return;
    
    List<SensorData> typeData = _dataBuffer
      .where((data) => data.type == dataType)
      .toList();
    
    if (typeData.isEmpty) return;
    
    // 计算平均值
    double average = typeData
      .map((data) => data.value)
      .reduce((a, b) => a + b) / typeData.length;
    
    // 计算趋势
    String trend = _calculateTrend(typeData);
    
    // 检测异常
    String status = _detectAnomalies(typeData, average);
    
    // 发送分析结果
    AnalysisResult result = AnalysisResult(
      metric: '${dataType}_average',
      value: average,
      status: '$status, $trend',
      timestamp: DateTime.now(),
    );
    
    _resultController.add(result);
  }
  
  String _calculateTrend(List<SensorData> data) {
    if (data.length < 2) return '趋势未知';
    
    double first = data.first.value;
    double last = data.last.value;
    double change = ((last - first) / first * 100);
    
    if (change > 5) return '上升趋势';
    if (change < -5) return '下降趋势';
    return '稳定趋势';
  }
  
  String _detectAnomalies(List<SensorData> data, double average) {
    SensorData latest = data.last;
    double deviation = (latest.value - average).abs();
    double threshold = average * 0.2; // 20%阈值
    
    if (deviation > threshold) {
      return '异常值检测';
    }
    return '正常范围';
  }
  
  void dispose() {
    _resultController.close();
  }
}

// 数据存储管理器
class DataStorageManager {
  final List<SensorData> _historicalData = [];
  final List<AnalysisResult> _analysisHistory = [];
  
  // 存储传感器数据
  void storeSensorData(SensorData data) {
    _historicalData.add(data);
    
    // 保持历史数据在合理范围内
    if (_historicalData.length > 1000) {
      _historicalData.removeAt(0);
    }
    
    print('💾 存储数据: ${data.sensorId} - ${data.value}');
  }
  
  // 存储分析结果
  void storeAnalysisResult(AnalysisResult result) {
    _analysisHistory.add(result);
    
    if (_analysisHistory.length > 500) {
      _analysisHistory.removeAt(0);
    }
    
    print('📊 存储分析: ${result.metric} - ${result.status}');
  }
  
  // 获取历史数据
  List<SensorData> getHistoricalData({
    String? sensorId,
    DateTime? startTime,
    DateTime? endTime,
  }) {
    return _historicalData.where((data) {
      bool matchSensor = sensorId == null || data.sensorId == sensorId;
      bool matchStart = startTime == null || data.timestamp.isAfter(startTime);
      bool matchEnd = endTime == null || data.timestamp.isBefore(endTime);
      
      return matchSensor && matchStart && matchEnd;
    }).toList();
  }
  
  // 生成统计报告
  Map<String, dynamic> generateStatistics() {
    Map<String, List<double>> typeValues = {};
    
    for (SensorData data in _historicalData) {
      typeValues.putIfAbsent(data.type, () => []);
      typeValues[data.type]!.add(data.value);
    }
    
    Map<String, dynamic> statistics = {};
    
    typeValues.forEach((type, values) {
      if (values.isNotEmpty) {
        double min = values.reduce((a, b) => a < b ? a : b);
        double max = values.reduce((a, b) => a > b ? a : b);
        double average = values.reduce((a, b) => a + b) / values.length;
        
        statistics[type] = {
          'count': values.length,
          'min': min,
          'max': max,
          'average': average,
        };
      }
    });
    
    return statistics;
  }
}

// 实时监控系统
class MonitoringSystem {
  final List<SensorSimulator> _sensors = [];
  final RealTimeAnalyzer _analyzer = RealTimeAnalyzer(bufferSize: 20);
  final DataStorageManager _storage = DataStorageManager();
  final List<StreamSubscription> _subscriptions = [];
  
  // 添加传感器
  void addSensor(String sensorId, String sensorType) {
    SensorSimulator sensor = SensorSimulator(sensorId, sensorType);
    _sensors.add(sensor);
    
    // 订阅传感器数据流
    StreamSubscription subscription = sensor.dataStream.listen(
      (data) {
        _analyzer.processSensorData(data);
        _storage.storeSensorData(data);
      },
      onError: (error) => print('❌ 传感器错误: $error'),
    );
    
    _subscriptions.add(subscription);
    print('✅ 添加传感器: $sensorId ($sensorType)');
  }
  
  // 开始监控
  void startMonitoring() {
    print('🚀 开始监控系统...');
    
    // 启动所有传感器
    for (SensorSimulator sensor in _sensors) {
      sensor.startGenerating(interval: Duration(milliseconds: 2000));
    }
    
    // 订阅分析结果
    StreamSubscription analysisSubscription = _analyzer.analysisResults.listen(
      (result) {
        _storage.storeAnalysisResult(result);
        print('📈 分析结果: $result');
        
        // 检查是否需要报警
        _checkAlerts(result);
      },
      onError: (error) => print('❌ 分析错误: $error'),
    );
    
    _subscriptions.add(analysisSubscription);
  }
  
  void _checkAlerts(AnalysisResult result) {
    if (result.status.contains('异常')) {
      print('🚨 警告: ${result.metric} 检测到异常值 ${result.value}');
    }
  }
  
  // 生成实时报告
  Future<void> generateRealTimeReport() async {
    Timer.periodic(Duration(seconds: 10), (timer) {
      print('\n📋 === 实时监控报告 ===');
      
      Map<String, dynamic> stats = _storage.generateStatistics();
      stats.forEach((type, data) {
        print('$type: 平均值=${data['average'].toStringAsFixed(2)}, '
              '最小值=${data['min']}, 最大值=${data['max']}, '
              '数据点=${data['count']}');
      });
      
      print('报告时间: ${DateTime.now()}');
      print('=' * 30);
    });
  }
  
  // 停止监控
  void stopMonitoring() {
    print('⏹️ 停止监控系统...');
    
    // 取消所有订阅
    for (StreamSubscription subscription in _subscriptions) {
      subscription.cancel();
    }
    
    // 停止所有传感器
    for (SensorSimulator sensor in _sensors) {
      sensor.stopGenerating();
    }
    
    _analyzer.dispose();
  }
}

// 主程序演示
Future<void> main() async {
  print('🎯 实时数据流处理系统演示');
  
  MonitoringSystem system = MonitoringSystem();
  
  // 添加多个传感器
  system.addSensor('TEMP_001', 'temperature');
  system.addSensor('HUM_001', 'humidity');
  system.addSensor('PRES_001', 'pressure');
  
  // 开始监控
  system.startMonitoring();
  
  // 生成实时报告
  system.generateRealTimeReport();
  
  // 运行30秒后停止
  await Future.delayed(Duration(seconds: 30));
  
  system.stopMonitoring();
  
  print('✨ 演示完成！');
}
```
