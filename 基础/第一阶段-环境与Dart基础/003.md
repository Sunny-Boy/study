### 1. Dart语言概述与特性

#### 1.1 Dart语言定位
Dart是Google开发的现代编程语言，专门为构建快速、可移植的应用而设计：

- **多平台**: 移动端、Web端、服务器端、桌面端
- **高性能**: 支持AOT和JIT两种编译模式
- **类型安全**: 强类型系统配合空安全机制
- **开发友好**: 简洁语法，丰富的标准库

#### 1.2 Dart核心特性

## 底层原理解析

### AOT vs JIT编译原理

Dart支持两种编译模式，适用于不同的场景：

```dart
// AOT (Ahead-Of-Time) 编译
/*
编译流程:
Dart Source → Dart Frontend → Kernel → AOT Compiler → Native Code

特点:
- 编译时将Dart代码转换为机器码
- 启动速度快，运行性能高
- 文件体积较大，编译时间长
- 用于生产环境（Release模式）

应用场景:
- Flutter Release构建
- 生产环境部署
- 性能要求高的场景
*/

// JIT (Just-In-Time) 编译
/*
编译流程:
Dart Source → Dart Frontend → Kernel → Dart VM → JIT Compiler → Machine Code

特点:
- 运行时动态编译
- 支持热重载和热重启
- 启动稍慢，但开发效率高
- 用于开发环境（Debug模式）

应用场景:
- Flutter开发调试
- Hot Reload功能
- 快速迭代开发
*/

class CompilationModeDemo {
  static void demonstrateCompilationDifference() {
    // 这段代码在不同编译模式下的行为：
    
    var startTime = DateTime.now();
    
    // 复杂计算
    int result = 0;
    for (int i = 0; i < 1000000; i++) {
      result += i * i;
    }
    
    var endTime = DateTime.now();
    var duration = endTime.difference(startTime);
    
    print('计算结果: $result');
    print('执行时间: ${duration.inMilliseconds}ms');
    
    // AOT模式：执行时间更短，性能更好
    // JIT模式：首次运行较慢，但支持动态特性
  }
}
```

/*
Dart VM架构组件:

1. Isolate（隔离区）
   - 独立的内存空间和执行上下文
   - 类似于操作系统进程，但更轻量

   - 自动内存管理
   - 分代回收算法
   - 减少停顿时间

// Isolate通信示例
import 'dart:isolate';

class IsolateDemo {
  static void spawnIsolateExample() async {
    // 创建新的Isolate
    ReceivePort receivePort = ReceivePort();
    
    await Isolate.spawn(isolateEntryPoint, receivePort.sendPort);
    
    // 监听Isolate消息
    receivePort.listen((data) {
      print('收到Isolate消息: $data');
    });
  }
  
  static void isolateEntryPoint(SendPort sendPort) {
    // 在新Isolate中执行的代码
    sendPort.send('Hello from Isolate!');
    
    // 执行耗时操作不会阻塞主Isolate
    int result = heavyComputation();
    sendPort.send('计算完成: $result');
  }
  
  static int heavyComputation() {
    int sum = 0;
    for (int i = 0; i < 10000000; i++) {
      sum += i;
    }
    return sum;
  }
}

## 大白话解释

把Dart语言想象成一个**智能翻译官**：

- **变量**就像是**标签盒子**，你可以往里面放不同的东西（数据）
- **类型**就像是**盒子的规格**，规定了能放什么样的东西
- **函数**就像是**专用工具**，输入原材料，输出加工好的产品
- **控制流**就像是**决策树**，根据不同条件走不同的路线

**AOT编译**就像是**提前做好的速冻食品**，加热就能吃，速度快但不能改口味。
**JIT编译**就像是**现场烹饪**，可以随时调整口味，但需要等待时间。

**空安全**就像是**保险箱机制**，明确标识哪些盒子可能是空的，避免打开空盒子时出现意外。

class Student {
  final String id;
  final String name;
  final Map<String, double> scores;
  
  Student({
    required this.id,
    required this.name,
    Map<String, double>? scores,
  }) : scores = scores ?? {};
  
  // 添加成绩
  void addScore(String subject, double score) {
    if (score < 0 || score > 100) {
      throw ArgumentError('成绩必须在0-100之间');
    }
    scores[subject] = score;
  }
  
  // 计算平均分
  double get averageScore {
    if (scores.isEmpty) return 0.0;
    double total = scores.values.reduce((a, b) => a + b);
    return total / scores.length;
  }
  
  // 获取等级
  String get grade {
    double avg = averageScore;
    if (avg >= 90) return 'A';
    if (avg >= 80) return 'B';  
    if (avg >= 70) return 'C';
    if (avg >= 60) return 'D';
    return 'F';
  }
  
  // 是否及格
  bool get isPassing => averageScore >= 60;
  
  @override
  String toString() {
    return '学生: $name (ID: $id), 平均分: ${averageScore.toStringAsFixed(2)}, 等级: $grade';
  }
}

// 成绩管理器
class GradeManager {
  final List<Student> _students = [];
  
  // 添加学生
  void addStudent(Student student) {
    // 检查ID是否重复
    if (_students.any((s) => s.id == student.id)) {
      throw ArgumentError('学生ID已存在: ${student.id}');
    }
    _students.add(student);
    print('✅ 学生 ${student.name} 添加成功');
  }
  
  // 根据ID查找学生
  Student? findStudentById(String id) {
    try {
      return _students.firstWhere((student) => student.id == id);
    } catch (e) {
      return null;
    }
  }
  
  // 添加成绩
  bool addGrade(String studentId, String subject, double score) {
    Student? student = findStudentById(studentId);
    if (student == null) {
      print('❌ 未找到学生ID: $studentId');
      return false;
    }
    
    try {
      student.addScore(subject, score);
      print('✅ ${student.name} 的 $subject 成绩已添加: $score');
      return true;
    } catch (e) {
      print('❌ 添加成绩失败: $e');
      return false;
    }
  }
  
  // 显示所有学生信息
  void displayAllStudents() {
    if (_students.isEmpty) {
      print('📝 暂无学生信息');
      return;
    }
    
    print('\n📊 学生成绩统计:');
    print('=' * 60);
    
    for (Student student in _students) {
      print(student);
      if (student.scores.isNotEmpty) {
        student.scores.forEach((subject, score) {
          print('  📚 $subject: $score');
        });
      } else {
        print('  📝 暂无成绩记录');
      }
      print('-' * 40);
    }
  }
  
  // 统计分析
  void generateStatistics() {
    if (_students.isEmpty) {
      print('📊 无数据可统计');
      return;
    }
    
    // 计算班级平均分
    double totalAverage = _students
      .map((student) => student.averageScore)
      .reduce((a, b) => a + b) / _students.length;
    
    // 找出最高分和最低分学生
    Student topStudent = _students
      .reduce((a, b) => a.averageScore > b.averageScore ? a : b);
    
    Student bottomStudent = _students
      .reduce((a, b) => a.averageScore < b.averageScore ? a : b);
    
    // 及格率统计
    int passingCount = _students.where((s) => s.isPassing).length;
    double passingRate = (passingCount / _students.length) * 100;
    
    print('\n📈 班级统计分析:');
    print('=' * 50);
    print('👥 总学生数: ${_students.length}');
    print('📊 班级平均分: ${totalAverage.toStringAsFixed(2)}');
    print('🏆 最高分学生: ${topStudent.name} (${topStudent.averageScore.toStringAsFixed(2)})');
    print('📉 最低分学生: ${bottomStudent.name} (${bottomStudent.averageScore.toStringAsFixed(2)})');
    print('✅ 及格人数: $passingCount');
    print('📊 及格率: ${passingRate.toStringAsFixed(1)}%');
  }
}

      '张三', '李四', '王五', '赵六', '陈七', 
      '刘八', '杨九', '黄十', '周一', '吴二'
    ];
    
    List<String> subjects = ['数学', '语文', '英语', '物理', '化学'];
    List<Student> students = [];
    
    for (int i = 0; i < 5; i++) {
      Student student = Student(
        id: 'STU${1001 + i}',
        name: names[i],
      );
      
      // 为每个学生添加随机成绩
      for (String subject in subjects) {
        double score = 60 + _random.nextDouble() * 40; // 60-100分
        student.addScore(subject, double.parse(score.toStringAsFixed(1)));
      }
      
      students.add(student);
    }
    
    return students;
  }
  
  // 输入验证
  static double? parseScore(String input) {
    try {
      double score = double.parse(input);
      if (score >= 0 && score <= 100) {
        return score;
      }
      return null;
    } catch (e) {
      return null;
    }
  }
}
