## MethodChannel双向通信

### 1. Flutter端实现
```dart
// lib/my_custom_plugin.dart
import 'dart:async';
import 'dart:typed_data';
import 'package:flutter/services.dart';

class MyCustomPlugin {
  static const MethodChannel _channel = MethodChannel('my_custom_plugin');

  // 获取平台版本
  static Future<String?> get platformVersion async {
    final String? version = await _channel.invokeMethod('getPlatformVersion');
    return version;
  }

  // 获取设备信息
  static Future<Map<String, dynamic>?> getDeviceInfo() async {
    try {
      final Map<String, dynamic>? deviceInfo = 
          await _channel.invokeMethod('getDeviceInfo');
      return deviceInfo;
    } on PlatformException catch (e) {
      print('获取设备信息失败: ${e.message}');
      return null;
    }
  }

  // 显示原生Toast (Android) / Alert (iOS)
  static Future<void> showNativeMessage(String message) async {
    await _channel.invokeMethod('showNativeMessage', {
      'message': message,
    });
  }

  // 计算两个数的和 (演示数据传递)
  static Future<int?> calculateSum(int a, int b) async {
    try {
      final int? result = await _channel.invokeMethod('calculateSum', {
        'a': a,
        'b': b,
      });
      return result;
    } on PlatformException catch (e) {
      print('计算失败: ${e.message}');
      return null;
    }
  }

  // 文件操作示例
  static Future<bool> saveFileToNative(String fileName, Uint8List data) async {
    try {
      final bool success = await _channel.invokeMethod('saveFile', {
        'fileName': fileName,
        'data': data,
      });
      return success;
    } on PlatformException catch (e) {
      print('保存文件失败: ${e.message}');
      return false;
    }
  }

  // 读取文件
  static Future<Uint8List?> readFileFromNative(String fileName) async {
    try {
      final Uint8List? data = await _channel.invokeMethod('readFile', {
        'fileName': fileName,
      });
      return data;
    } on PlatformException catch (e) {
      print('读取文件失败: ${e.message}');
      return null;
    }
  }

  // 异步操作示例
  static Future<String?> performLongTask(String taskName) async {
    try {
      final String? result = await _channel.invokeMethod('performLongTask', {
        'taskName': taskName,
      });
      return result;
    } on PlatformException catch (e) {
      print('执行长任务失败: ${e.message}');
      return null;
    }
  }
}
```

### 2. Android端实现
```java
// android/src/main/java/com/example/my_custom_plugin/MyCustomPlugin.java
package com.example.my_custom_plugin;

import android.content.Context;
import android.content.pm.PackageInfo;
import android.content.pm.PackageManager;
import android.os.AsyncTask;
import android.os.Build;
import android.widget.Toast;

import androidx.annotation.NonNull;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.IOException;
import java.util.HashMap;
import java.util.Map;

import io.flutter.embedding.engine.plugins.FlutterPlugin;
import io.flutter.plugin.common.MethodCall;
import io.flutter.plugin.common.MethodChannel;
import io.flutter.plugin.common.MethodChannel.MethodCallHandler;
import io.flutter.plugin.common.MethodChannel.Result;

public class MyCustomPlugin implements FlutterPlugin, MethodCallHandler {
  private MethodChannel channel;
  private Context context;

  @Override
  public void onAttachedToEngine(@NonNull FlutterPluginBinding flutterPluginBinding) {
    channel = new MethodChannel(flutterPluginBinding.getBinaryMessenger(), "my_custom_plugin");
    channel.setMethodCallHandler(this);
    context = flutterPluginBinding.getApplicationContext();
  }

  @Override
  public void onMethodCall(@NonNull MethodCall call, @NonNull Result result) {
    switch (call.method) {
      case "getPlatformVersion":
        result.success("Android " + Build.VERSION.RELEASE);
        break;
      
      case "getDeviceInfo":
        getDeviceInfo(result);
        break;
      
      case "showNativeMessage":
        String message = call.argument("message");
        showToast(message);
        result.success(null);
        break;
      
      case "calculateSum":
        Integer a = call.argument("a");
        Integer b = call.argument("b");
        if (a != null && b != null) {
          result.success(a + b);
        } else {
          result.error("INVALID_ARGUMENTS", "参数不能为空", null);
        }
        break;
      
      case "saveFile":
        String fileName = call.argument("fileName");
        byte[] data = call.argument("data");
        saveFile(fileName, data, result);
        break;
      
      case "readFile":
        String readFileName = call.argument("fileName");
        readFile(readFileName, result);
        break;
      
      case "performLongTask":
        String taskName = call.argument("taskName");
        performLongTask(taskName, result);
        break;
      
      default:
        result.notImplemented();
        break;
    }
  }

  private void getDeviceInfo(Result result) {
    try {
      Map<String, Object> deviceInfo = new HashMap<>();
      deviceInfo.put("model", Build.MODEL);
      deviceInfo.put("manufacturer", Build.MANUFACTURER);
      deviceInfo.put("version", Build.VERSION.RELEASE);
      deviceInfo.put("sdkVersion", Build.VERSION.SDK_INT);
      deviceInfo.put("brand", Build.BRAND);
      
      // 应用信息
      PackageManager pm = context.getPackageManager();
      PackageInfo packageInfo = pm.getPackageInfo(context.getPackageName(), 0);
      deviceInfo.put("appVersion", packageInfo.versionName);
      deviceInfo.put("packageName", context.getPackageName());
      
      result.success(deviceInfo);
    } catch (Exception e) {
      result.error("DEVICE_INFO_ERROR", "获取设备信息失败", e.getMessage());
    }
  }

  private void showToast(String message) {
    if (message != null) {
      Toast.makeText(context, message, Toast.LENGTH_SHORT).show();
    }
  }

  private void saveFile(String fileName, byte[] data, Result result) {
    if (fileName == null || data == null) {
      result.error("INVALID_ARGUMENTS", "文件名和数据不能为空", null);
      return;
    }

    try {
      File file = new File(context.getFilesDir(), fileName);
      FileOutputStream fos = new FileOutputStream(file);
      fos.write(data);
      fos.close();
      result.success(true);
    } catch (IOException e) {
      result.error("FILE_SAVE_ERROR", "保存文件失败", e.getMessage());
    }
  }

  private void readFile(String fileName, Result result) {
    if (fileName == null) {
      result.error("INVALID_ARGUMENTS", "文件名不能为空", null);
      return;
    }

    try {
      File file = new File(context.getFilesDir(), fileName);
      if (!file.exists()) {
        result.error("FILE_NOT_EXISTS", "文件不存在", null);
        return;
      }

      FileInputStream fis = new FileInputStream(file);
      byte[] data = new byte[(int) file.length()];
      fis.read(data);
      fis.close();
      result.success(data);
    } catch (IOException e) {
      result.error("FILE_READ_ERROR", "读取文件失败", e.getMessage());
    }
  }

  private void performLongTask(String taskName, Result result) {
    new AsyncTask<Void, Void, String>() {
      @Override
      protected String doInBackground(Void... voids) {
        try {
          // 模拟长时间任务
          Thread.sleep(3000);
          return "任务 '" + taskName + "' 执行完成";
        } catch (InterruptedException e) {
          return "任务被中断";
        }
      }

      @Override
      protected void onPostExecute(String s) {
        result.success(s);
      }
    }.execute();
  }

  @Override
  public void onDetachedFromEngine(@NonNull FlutterPluginBinding binding) {
    channel.setMethodCallHandler(null);
  }
}
```

```objective-c
// ios/Classes/MyCustomPlugin.m
#import "MyCustomPlugin.h"
#import <UIKit/UIKit.h>

@implementation MyCustomPlugin
+ (void)registerWithRegistrar:(NSObject<FlutterPluginRegistrar>*)registrar {
  FlutterMethodChannel* channel = [FlutterMethodChannel
      methodChannelWithName:@"my_custom_plugin"
            binaryMessenger:[registrar messenger]];
  MyCustomPlugin* instance = [[MyCustomPlugin alloc] init];
  [registrar addMethodCallDelegate:instance channel:channel];
}

- (void)handleMethodCall:(FlutterMethodCall*)call result:(FlutterResult)result {
  if ([@"getPlatformVersion" isEqualToString:call.method]) {
    result([@"iOS " stringByAppendingString:[[UIDevice currentDevice] systemVersion]]);
  } else if ([@"getDeviceInfo" isEqualToString:call.method]) {
    [self getDeviceInfo:result];
  } else if ([@"showNativeMessage" isEqualToString:call.method]) {
    NSString* message = call.arguments[@"message"];
    [self showAlert:message];
    result(nil);
  } else if ([@"calculateSum" isEqualToString:call.method]) {
    NSNumber* a = call.arguments[@"a"];
    NSNumber* b = call.arguments[@"b"];
    if (a && b) {
      NSInteger sum = [a integerValue] + [b integerValue];
      result(@(sum));
    } else {
      result([FlutterError errorWithCode:@"INVALID_ARGUMENTS"
                                 message:@"参数不能为空"
                                 details:nil]);
    }
  } else if ([@"saveFile" isEqualToString:call.method]) {
    NSString* fileName = call.arguments[@"fileName"];
    FlutterStandardTypedData* data = call.arguments[@"data"];
    [self saveFile:fileName data:data.data result:result];
  } else if ([@"readFile" isEqualToString:call.method]) {
    NSString* fileName = call.arguments[@"fileName"];
    [self readFile:fileName result:result];
  } else if ([@"performLongTask" isEqualToString:call.method]) {
    NSString* taskName = call.arguments[@"taskName"];
    [self performLongTask:taskName result:result];
  } else {
    result(FlutterMethodNotImplemented);
  }
}

- (void)getDeviceInfo:(FlutterResult)result {
  UIDevice* device = [UIDevice currentDevice];
  NSBundle* bundle = [NSBundle mainBundle];
  
  NSDictionary* deviceInfo = @{
    @"model": device.model,
    @"name": device.name,
    @"systemName": device.systemName,
    @"systemVersion": device.systemVersion,
    @"identifierForVendor": [device.identifierForVendor UUIDString] ?: @"",
    @"appVersion": [bundle objectForInfoDictionaryKey:@"CFBundleShortVersionString"] ?: @"",
    @"bundleId": bundle.bundleIdentifier ?: @""
  };
  
  result(deviceInfo);
}

- (void)showAlert:(NSString*)message {
  if (!message) return;
  
  dispatch_async(dispatch_get_main_queue(), ^{
    UIAlertController* alert = [UIAlertController
        alertControllerWithTitle:@"提示"
                         message:message
                  preferredStyle:UIAlertControllerStyleAlert];
    
    UIAlertAction* okAction = [UIAlertAction
        actionWithTitle:@"确定"
                  style:UIAlertActionStyleDefault
                handler:nil];
    
    [alert addAction:okAction];
    
    UIViewController* rootViewController = [UIApplication sharedApplication].keyWindow.rootViewController;
    [rootViewController presentViewController:alert animated:YES completion:nil];
  });
}

- (void)saveFile:(NSString*)fileName data:(NSData*)data result:(FlutterResult)result {
  if (!fileName || !data) {
    result([FlutterError errorWithCode:@"INVALID_ARGUMENTS"
                               message:@"文件名和数据不能为空"
                               details:nil]);
    return;
  }
  
  NSArray* paths = NSSearchPathForDirectoriesInDomains(NSDocumentDirectory, NSUserDomainMask, YES);
  NSString* documentsDirectory = [paths objectAtIndex:0];
  NSString* filePath = [documentsDirectory stringByAppendingPathComponent:fileName];
  
  BOOL success = [data writeToFile:filePath atomically:YES];
  result(@(success));
}

- (void)readFile:(NSString*)fileName result:(FlutterResult)result {
  if (!fileName) {
    result([FlutterError errorWithCode:@"INVALID_ARGUMENTS"
                               message:@"文件名不能为空"
                               details:nil]);
    return;
  }
  
  NSArray* paths = NSSearchPathForDirectoriesInDomains(NSDocumentDirectory, NSUserDomainMask, YES);
  NSString* documentsDirectory = [paths objectAtIndex:0];
  NSString* filePath = [documentsDirectory stringByAppendingPathComponent:fileName];
  
  NSData* data = [NSData dataWithContentsOfFile:filePath];
  if (data) {
    FlutterStandardTypedData* typedData = [FlutterStandardTypedData typedDataWithBytes:data];
    result(typedData);
  } else {
    result([FlutterError errorWithCode:@"FILE_NOT_EXISTS"
                               message:@"文件不存在"
                               details:nil]);
  }
}

- (void)performLongTask:(NSString*)taskName result:(FlutterResult)result {
  dispatch_async(dispatch_get_global_queue(DISPATCH_QUEUE_PRIORITY_DEFAULT, 0), ^{
    // 模拟长时间任务
    [NSThread sleepForTimeInterval:3.0];
    
    NSString* resultMessage = [NSString stringWithFormat:@"任务 '%@' 执行完成", taskName];
    
    dispatch_async(dispatch_get_main_queue(), ^{
      result(resultMessage);
    });
  });
}
```

```dart
// lib/my_custom_plugin_events.dart
import 'dart:async';
import 'package:flutter/services.dart';

class MyCustomPluginEvents {
  static const EventChannel _eventChannel = EventChannel('my_custom_plugin/events');
  
  // 监听传感器数据
  static Stream<Map<String, dynamic>> get sensorDataStream {
    return _eventChannel.receiveBroadcastStream('sensor_data')
        .map<Map<String, dynamic>>((dynamic event) => 
            Map<String, dynamic>.from(event));
  }
  
  // 监听网络状态变化
  static Stream<String> get networkStatusStream {
    return _eventChannel.receiveBroadcastStream('network_status')
        .map<String>((dynamic event) => event.toString());
  }
  
  // 监听电池状态
  static Stream<Map<String, dynamic>> get batteryStatusStream {
    return _eventChannel.receiveBroadcastStream('battery_status')
        .map<Map<String, dynamic>>((dynamic event) => 
            Map<String, dynamic>.from(event));
  }
}

// 使用示例
class SensorWidget extends StatefulWidget {
  @override
  _SensorWidgetState createState() => _SensorWidgetState();
}

class _SensorWidgetState extends State<SensorWidget> {
  StreamSubscription<Map<String, dynamic>>? _sensorSubscription;
  String _sensorData = '暂无数据';
  
  @override
  void initState() {
    super.initState();
    _startListeningSensorData();
  }
  
  void _startListeningSensorData() {
    _sensorSubscription = MyCustomPluginEvents.sensorDataStream.listen(
      (data) {
        setState(() {
          _sensorData = '加速度: ${data['accelerometer']}\n'
                       '陀螺仪: ${data['gyroscope']}\n'
                       '磁力计: ${data['magnetometer']}';
        });
      },
      onError: (error) {
        print('传感器数据监听错误: $error');
      },
    );
  }
  
  @override
  void dispose() {
    _sensorSubscription?.cancel();
    super.dispose();
  }
  
  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        Text('传感器数据:'),
        Text(_sensorData),
      ],
    );
  }
}
```

### 2. Android端EventChannel实现
```java
// Android EventChannel 处理器
package com.example.my_custom_plugin;

import android.content.Context;
import android.hardware.Sensor;
import android.hardware.SensorEvent;
import android.hardware.SensorEventListener;
import android.hardware.SensorManager;
import android.os.Handler;
import android.os.Looper;

import java.util.HashMap;
import java.util.Map;

import io.flutter.plugin.common.EventChannel;

public class SensorEventHandler implements EventChannel.StreamHandler, SensorEventListener {
    private Context context;
    private SensorManager sensorManager;
    private Sensor accelerometer;
    private Sensor gyroscope;
    private Sensor magnetometer;
    private EventChannel.EventSink eventSink;
    private Handler mainHandler;

    public SensorEventHandler(Context context) {
        this.context = context;
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.accelerometer = sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER);
        this.gyroscope = sensorManager.getDefaultSensor(Sensor.TYPE_GYROSCOPE);
        this.magnetometer = sensorManager.getDefaultSensor(Sensor.TYPE_MAGNETIC_FIELD);
        this.mainHandler = new Handler(Looper.getMainLooper());
    }

    @Override
    public void onListen(Object arguments, EventChannel.EventSink events) {
        this.eventSink = events;
        
        if (accelerometer != null) {
            sensorManager.registerListener(this, accelerometer, SensorManager.SENSOR_DELAY_NORMAL);
        }
        if (gyroscope != null) {
            sensorManager.registerListener(this, gyroscope, SensorManager.SENSOR_DELAY_NORMAL);
        }
        if (magnetometer != null) {
            sensorManager.registerListener(this, magnetometer, SensorManager.SENSOR_DELAY_NORMAL);
        }
    }

    @Override
    public void onCancel(Object arguments) {
        sensorManager.unregisterListener(this);
        this.eventSink = null;
    }

    @Override
    public void onSensorChanged(SensorEvent event) {
        if (eventSink == null) return;

        Map<String, Object> sensorData = new HashMap<>();
        
        switch (event.sensor.getType()) {
            case Sensor.TYPE_ACCELEROMETER:
                Map<String, Float> accelerometerData = new HashMap<>();
                accelerometerData.put("x", event.values[0]);
                accelerometerData.put("y", event.values[1]);
                accelerometerData.put("z", event.values[2]);
                sensorData.put("accelerometer", accelerometerData);
                break;
                
            case Sensor.TYPE_GYROSCOPE:
                Map<String, Float> gyroscopeData = new HashMap<>();
                gyroscopeData.put("x", event.values[0]);
                gyroscopeData.put("y", event.values[1]);
                gyroscopeData.put("z", event.values[2]);
                sensorData.put("gyroscope", gyroscopeData);
                break;
                
            case Sensor.TYPE_MAGNETIC_FIELD:
                Map<String, Float> magnetometerData = new HashMap<>();
                magnetometerData.put("x", event.values[0]);
                magnetometerData.put("y", event.values[1]);
                magnetometerData.put("z", event.values[2]);
                sensorData.put("magnetometer", magnetometerData);
                break;
        }

        // 确保在主线程发送数据
        mainHandler.post(() -> eventSink.success(sensorData));
    }

    @Override
    public void onAccuracyChanged(Sensor sensor, int accuracy) {
        // 传感器精度变化处理
    }
}

// 在主插件类中注册EventChannel
public class MyCustomPlugin implements FlutterPlugin, MethodCallHandler {
    private MethodChannel methodChannel;
    private EventChannel eventChannel;
    private SensorEventHandler sensorEventHandler;

    @Override
    public void onAttachedToEngine(@NonNull FlutterPluginBinding flutterPluginBinding) {
        methodChannel = new MethodChannel(flutterPluginBinding.getBinaryMessenger(), "my_custom_plugin");
        methodChannel.setMethodCallHandler(this);
        
        eventChannel = new EventChannel(flutterPluginBinding.getBinaryMessenger(), "my_custom_plugin/events");
        sensorEventHandler = new SensorEventHandler(flutterPluginBinding.getApplicationContext());
        eventChannel.setStreamHandler(sensorEventHandler);
    }

    @Override
    public void onDetachedFromEngine(@NonNull FlutterPluginBinding binding) {
        methodChannel.setMethodCallHandler(null);
        eventChannel.setStreamHandler(null);
    }
}
```

### 2. 错误处理与调试
```dart
class PluginDebugger {
  static const MethodChannel _channel = MethodChannel('plugin_debugger');

  static Future<T?> safeInvoke<T>(
    String method, 
    dynamic arguments, {
    T? defaultValue,
    bool logErrors = true,
  }) async {
    try {
      final result = await _channel.invokeMethod<T>(method, arguments);
      return result ?? defaultValue;
    } on PlatformException catch (e) {
      if (logErrors) {
        print('插件调用失败: $method');
        print('错误代码: ${e.code}');
        print('错误信息: ${e.message}');
        print('错误详情: ${e.details}');
      }
      return defaultValue;
    } catch (e) {
      if (logErrors) {
        print('未知错误: $e');
      }
      return defaultValue;
    }
  }
}

// 使用示例

  'getDeviceInfo',
  null,
  defaultValue: <String, dynamic>{},
```

```dart
  test('计算两数之和', () async {
```
