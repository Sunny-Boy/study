### 3. Hot Reload和Hot Restart机制

#### 3.1 Hot Reload工作原理

```dart
/*
Hot Reload工作流程:

1. 文件监听 → 2. 增量编译 → 3. 代码注入 → 4. Widget重建

详细过程:
┌─────────────────┐    ┌─────────────────┐    ┌─────────────────┐
│   文件系统监听   │ ─→ │   Dart编译器     │ ─→ │   Flutter Engine │
│   检测代码变更   │    │   增量编译      │    │   更新应用状态   │
└─────────────────┘    └─────────────────┘    └─────────────────┘
         ↓                       ↓                       ↓
┌─────────────────┐    ┌─────────────────┐    ┌─────────────────┐
│   IDE/编辑器     │    │   VM Service    │    │   Widget Tree   │
│   保存代码      │    │   代码热注入    │    │   重建UI        │
└─────────────────┘    └─────────────────┘    └─────────────────┘
*/

// Hot Reload支持的变更类型
class HotReloadExamples {
  // ✅ 支持Hot Reload的变更
  static void supportedChanges() {
    // 1. Widget的build方法修改
    Widget buildExample() {
      return Container(
        color: Colors.blue, // 可以热重载修改颜色
        child: Text('Hello'), // 可以热重载修改文本
      );
    }
    
    // 2. 方法实现修改
    void methodExample() {
      print('方法内容可以热重载'); // 可以修改打印内容
    }
    
    // 3. 常量值修改
    final String title = 'App Title'; // 可以修改字符串值
    final int count = 10; // 可以修改数值
  }
  
  // ❌ 不支持Hot Reload的变更（需要Hot Restart）
  static void unsupportedChanges() {
    // 1. 类的结构变更
    // - 添加/删除字段
    // - 修改构造函数签名
    // - 添加/删除方法
    
    // 2. 全局变量和静态字段
    // static String globalVar = 'value';
    
    // 3. main()函数修改
    // void main() {} 
    
    // 4. 枚举类型修改
    // enum Color { red, green, blue }
    
    // 5. 泛型类型变更
    // class GenericClass<T> {}
  }
}
```

#### 3.2 Hot Reload最佳实践

```dart
// 设计Hot Reload友好的代码结构
class HotReloadBestPractices {
  
  // ✅ 好的实践：将UI逻辑分离到build方法中
  class GoodWidget extends StatefulWidget {
    @override
    _GoodWidgetState createState() => _GoodWidgetState();
  }
  
  class _GoodWidgetState extends State<GoodWidget> {
    int counter = 0;
    
    @override
    Widget build(BuildContext context) {
      return Scaffold(
        appBar: AppBar(
          title: Text('计数器应用'), // 可以热重载修改
          backgroundColor: Colors.blue, // 可以热重载修改
        ),
        body: Center(
          child: Column(
            children: [
              Text(
                '当前计数: $counter',
                style: TextStyle(fontSize: 24), // 可以热重载修改
              ),
              SizedBox(height: 20), // 可以热重载修改间距
              ElevatedButton(
                onPressed: () => setState(() => counter++),
                child: Text('增加'), // 可以热重载修改
              ),
            ],
          ),
        ),
      );
    }
  }
  
  // ❌ 不好的实践：在构造函数中初始化复杂对象
  class BadWidget extends StatefulWidget {
    final ComplexObject complexObject;
    
    BadWidget() : complexObject = ComplexObject() {
      // 构造函数中的逻辑不支持热重载
      complexObject.initialize();
    }
    
    @override
    _BadWidgetState createState() => _BadWidgetState();
  }
  
  // ✅ 改进：将初始化移到initState中
  class ImprovedWidget extends StatefulWidget {
    @override
    _ImprovedWidgetState createState() => _ImprovedWidgetState();
  }
  
  class _ImprovedWidgetState extends State<ImprovedWidget> {
    late ComplexObject complexObject;
    
    @override
    void initState() {
      super.initState();
      complexObject = ComplexObject(); // 状态会在热重载时保留
      complexObject.initialize();
    }
    
    @override
    Widget build(BuildContext context) {
      // UI逻辑可以热重载
      return Container();
    }
  }
}

class ComplexObject {
  void initialize() {
    print('复杂对象初始化');
  }
}
```

#### 3.3 调试Hot Reload问题

```dart
// Hot Reload问题诊断和解决
class HotReloadTroubleshooting {
  
  // 问题1：状态没有正确更新
  class StateProblem extends StatefulWidget {
    @override
    _StateProblemState createState() => _StateProblemState();
  }
  
  class _StateProblemState extends State<StateProblem> {
    // ❌ 问题：全局变量不会热重载
    static int globalCounter = 0;
    
    // ✅ 解决：使用实例变量
    int instanceCounter = 0;
    
    @override
    Widget build(BuildContext context) {
      return Column(
        children: [
          Text('全局计数: $globalCounter'), // 不会热重载更新
          Text('实例计数: $instanceCounter'), // 会热重载更新
        ],
      );
    }
  }
  
  // 问题2：const构造函数导致的缓存问题
  class ConstProblem extends StatelessWidget {
    @override
    Widget build(BuildContext context) {
      return Column(
        children: [
          // ❌ 问题：const构造函数会缓存Widget
          const Text('不变的文本'),
          
          // ✅ 解决：去掉const或使用变量
          Text('可变的文本'),
        ],
      );
    }
  }
  
  // 问题3：闭包捕获的问题
  class ClosureProblem extends StatefulWidget {
    @override
    _ClosureProblemState createState() => _ClosureProblemState();
  }
  
  class _ClosureProblemState extends State<ClosureProblem> {
    String message = '初始消息';
    
    @override
    Widget build(BuildContext context) {
      // ❌ 问题：闭包捕获旧值
      VoidCallback badCallback = () {
        print(message); // 可能捕获旧值
      };
      
      // ✅ 解决：使用方法引用或在闭包内访问
      VoidCallback goodCallback = () {
        print(this.message); // 始终访问最新值
      };
      
      return ElevatedButton(
        onPressed: goodCallback,
        child: Text(message),
      );
    }
  }
}
```

```dart
  late Animation<double> _animation;
  bool _isExpanded = false;

    super.initState();

      duration: Duration(milliseconds: 500),
      vsync: this,
    );
    
    _animation = Tween<double>(
      begin: 0.0,
      end: 1.0,
    ).animate(CurvedAnimation(
      parent: _controller,
      curve: Curves.easeInOut,
    ));
  }
  
  @override
  Widget build(BuildContext context) {
    // 使用Inspector调试技巧：
    // 1. 在Widget上右键选择"Inspect Widget"
```

```dart
// lib/core/services/app_service.dart - 核心服务
import 'package:shared_preferences/shared_preferences.dart';

class AppService {
  late SharedPreferences _prefs;
  
  Future<void> initialize() async {
    _prefs = await SharedPreferences.getInstance();
    print('📱 应用服务初始化完成');
  }
  
  // 获取用户偏好
  T? getPreference<T>(String key) {
    switch (T) {
      case String:
        return _prefs.getString(key) as T?;
      case int:
        return _prefs.getInt(key) as T?;
      case bool:
        return _prefs.getBool(key) as T?;
      default:
        return null;
    }
  }
  
  // 设置用户偏好
  Future<bool> setPreference<T>(String key, T value) async {
    switch (T) {
      case String:
        return await _prefs.setString(key, value as String);
      case int:
        return await _prefs.setInt(key, value as int);
      case bool:
        return await _prefs.setBool(key, value as bool);
      default:
        return false;
    }
  }
}
```
