# 014 - Provider 状态管理

## 概述

Provider 是 Flutter 官方推荐的状态管理方案，基于 InheritedWidget 构建，提供了简洁、高效的状态共享和依赖注入机制。本章将深入学习 Provider 的架构设计、核心组件以及实际应用。

## Provider 架构设计

### 核心概念

Provider 采用了发布-订阅模式，通过 ChangeNotifier 和 Consumer 实现状态的发布和监听。

```dart
// 首先添加依赖
// pubspec.yaml
// dependencies:
//   provider: ^6.0.5

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

// 状态模型
class Counter extends ChangeNotifier {
  int _count = 0;

  int get count => _count;

  void increment() {
    _count++;
    notifyListeners(); // 通知监听者状态改变
  }

  void decrement() {
    _count--;
    notifyListeners();
  }

  void reset() {
    _count = 0;
    notifyListeners();
  }
}

// 应用入口
class CounterApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return ChangeNotifierProvider(
      create: (context) => Counter(),
      child: MaterialApp(
        title: 'Provider 计数器',
        home: CounterScreen(),
      ),
    );
  }
}
```

```dart
class UserProvider extends ChangeNotifier {
  User? _user;
  bool _isLoading = false;

  User? get user => _user;
  bool get isLoading => _isLoading;

  Future<void> login(String username, String password) async {
    _isLoading = true;
    notifyListeners();

    try {
      // 模拟网络请求
      await Future.delayed(Duration(seconds: 2));
      _user = User(username: username, email: '$username@example.com');
    } catch (e) {
      print('登录失败: $e');
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  void logout() {
    _user = null;
    notifyListeners();
  }
}

class ThemeNotifier extends ValueNotifier<ThemeMode> {
```

## ChangeNotifier 详解

### ChangeNotifier 基础用法

```dart
class TodosProvider extends ChangeNotifier {
  List<Todo> _todos = [];
  String _filter = 'all';

  List<Todo> get todos => _todos;
  String get filter => _filter;

  List<Todo> get filteredTodos {
    switch (_filter) {
      case 'active':
        return _todos.where((todo) => !todo.completed).toList();
      case 'completed':
        return _todos.where((todo) => todo.completed).toList();
      default:
        return _todos;
    }
  }

  int get activeCount => _todos.where((todo) => !todo.completed).length;
  int get completedCount => _todos.where((todo) => todo.completed).length;

  void addTodo(String title) {
    if (title.trim().isEmpty) return;
    
    _todos.add(Todo(
      id: DateTime.now().millisecondsSinceEpoch,
      title: title.trim(),
      completed: false,
    ));
    notifyListeners();
  }

  void toggleTodo(int id) {
    final index = _todos.indexWhere((todo) => todo.id == id);
    if (index != -1) {
      _todos[index] = _todos[index].copyWith(
        completed: !_todos[index].completed,
      );
      notifyListeners();
    }
  }

  void deleteTodo(int id) {
    _todos.removeWhere((todo) => todo.id == id);
    notifyListeners();
  }

  void updateTodo(int id, String newTitle) {
    final index = _todos.indexWhere((todo) => todo.id == id);
    if (index != -1) {
      _todos[index] = _todos[index].copyWith(title: newTitle);
      notifyListeners();
    }
  }

  void setFilter(String newFilter) {
    if (_filter != newFilter) {
      _filter = newFilter;
      notifyListeners();
    }
  }

  void clearCompleted() {
    _todos.removeWhere((todo) => todo.completed);
    notifyListeners();
  }

  void toggleAll() {
    final allCompleted = _todos.every((todo) => todo.completed);
    for (var todo in _todos) {
      _todos[_todos.indexOf(todo)] = todo.copyWith(completed: !allCompleted);
    }
    notifyListeners();
  }
}

class Todo {
  final int id;
  final String title;
  final bool completed;

  Todo({
    required this.id,
    required this.title,
    required this.completed,
  });

  Todo copyWith({String? title, bool? completed}) {
    return Todo(
      id: id,
      title: title ?? this.title,
      completed: completed ?? this.completed,
    );
  }

  @override
  bool operator ==(Object other) {
    if (identical(this, other)) return true;
    return other is Todo &&
        other.id == id &&
        other.title == title &&
        other.completed == completed;
  }

  @override
  int get hashCode => id.hashCode ^ title.hashCode ^ completed.hashCode;
}
```

### 高级 ChangeNotifier 模式

```dart
// 1. 延迟通知模式
class BatchUpdateProvider extends ChangeNotifier {
  List<String> _items = [];
  bool _isUpdating = false;

  List<String> get items => List.unmodifiable(_items);

  // 批量更新，减少通知次数
  void batchUpdate(List<String> newItems) {
    if (_isUpdating) return;
    
    _isUpdating = true;
    _items.clear();
    _items.addAll(newItems);
    _isUpdating = false;
    
    notifyListeners();
  }

  void addItem(String item) {
    _items.add(item);
    if (!_isUpdating) {
      notifyListeners();
    }
  }
}

// 2. 条件通知模式
class SmartNotificationProvider extends ChangeNotifier {
  String _status = '';
  DateTime _lastUpdate = DateTime.now();

  String get status => _status;
  DateTime get lastUpdate => _lastUpdate;

  void updateStatus(String newStatus) {
    // 只在状态真正改变时通知
    if (_status != newStatus) {
      _status = newStatus;
      _lastUpdate = DateTime.now();
      notifyListeners();
    }
  }

  // 静默更新，不触发通知
  void silentUpdate(String newStatus) {
    _status = newStatus;
    _lastUpdate = DateTime.now();
    // 不调用 notifyListeners()
  }
}

// 3. 异步操作处理
class AsyncDataProvider extends ChangeNotifier {
  String? _data;
  bool _isLoading = false;
  String? _error;

  String? get data => _data;
  bool get isLoading => _isLoading;
  String? get error => _error;
  bool get hasError => _error != null;

  Future<void> loadData() async {
    if (_isLoading) return; // 防止重复请求

    _isLoading = true;
    _error = null;
    notifyListeners();

    try {
      // 模拟网络请求
      await Future.delayed(Duration(seconds: 2));
      
      // 模拟随机失败
      if (DateTime.now().millisecond % 3 == 0) {
        throw Exception('网络错误');
      }
      
      _data = '加载的数据: ${DateTime.now()}';
    } catch (e) {
      _error = e.toString();
      _data = null;
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  void clearError() {
    if (_error != null) {
      _error = null;
      notifyListeners();
    }
  }
}
```

### Consumer 基础用法

```dart
class ConsumerExample extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Consumer 示例')),
      body: Column(
        children: [
          // Consumer - 监听整个 Provider
          Consumer<TodosProvider>(
            builder: (context, todosProvider, child) {
              return Card(
                child: ListTile(
                  title: Text('总任务数: ${todosProvider.todos.length}'),
                  subtitle: Text(
                    '活跃: ${todosProvider.activeCount}, '
                    '完成: ${todosProvider.completedCount}',
                  ),
                ),
              );
            },
          ),

          // Consumer 与 child 优化
          Consumer<TodosProvider>(
            child: Icon(Icons.list, size: 48), // 不会重建的部分
            builder: (context, todosProvider, child) {
              return Card(
                child: Column(
                  children: [
                    child!, // 复用不变的 child
                    Text('当前过滤器: ${todosProvider.filter}'),
                  ],
                ),
              );
            },
          ),

          // Consumer2 - 监听两个 Provider
          Consumer2<TodosProvider, UserProvider>(
            builder: (context, todosProvider, userProvider, child) {
              return Card(
                child: ListTile(
                  title: Text('用户: ${userProvider.user?.username ?? "未登录"}'),
                  subtitle: Text('任务数: ${todosProvider.todos.length}'),
                ),
              );
            },
          ),
        ],
      ),
    );
  }
}
```

```dart
class SelectorExample extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Selector 示例')),
      body: Column(
        children: [
          // Selector - 只监听特定属性
          Selector<TodosProvider, int>(
            selector: (context, provider) => provider.activeCount,
            builder: (context, activeCount, child) {
              print('重建活跃计数器'); // 调试用
              return Card(
                child: ListTile(
                  leading: Icon(Icons.pending_actions),
                  title: Text('活跃任务'),
                  trailing: Text('$activeCount'),
                ),
              );
            },
          ),

          // Selector 与 shouldRebuild
          Selector<TodosProvider, String>(
            selector: (context, provider) => provider.filter,
            shouldRebuild: (previous, next) {
              print('过滤器变化: $previous -> $next');
              return previous != next;
            },
            builder: (context, filter, child) {
              return Card(
                child: ListTile(
                  leading: Icon(Icons.filter_list),
                  title: Text('当前过滤器'),
                  trailing: Text(filter),
                ),
              );
            },
          ),

          // 复杂 Selector - 监听多个属性
          Selector<TodosProvider, ({int total, int active, int completed})>(
            selector: (context, provider) => (
              total: provider.todos.length,
              active: provider.activeCount,
              completed: provider.completedCount,
            ),
            builder: (context, stats, child) {
              return Card(
                child: Padding(

                    ],

        ],
```

```dart
    required this.name,

// 购物车项目
class CartItem {
  final Product product;
  final int quantity;

  CartItem({required this.product, required this.quantity});

  CartItem copyWith({Product? product, int? quantity}) {
    return CartItem(
      product: product ?? this.product,
      quantity: quantity ?? this.quantity,
    );
  }

  double get totalPrice => product.price * quantity;
}

// 购物车状态管理
class ShoppingCartProvider extends ChangeNotifier {
  final List<CartItem> _items = [];
  final List<Product> _favorites = [];

  List<CartItem> get items => List.unmodifiable(_items);
  List<Product> get favorites => List.unmodifiable(_favorites);

  int get itemCount => _items.fold(0, (sum, item) => sum + item.quantity);
  double get totalPrice => _items.fold(0.0, (sum, item) => sum + item.totalPrice);

  bool isInCart(int productId) {
    return _items.any((item) => item.product.id == productId);
  }

  bool isFavorite(int productId) {
    return _favorites.any((product) => product.id == productId);
  }

  void addToCart(Product product) {
    final existingIndex = _items.indexWhere(
      (item) => item.product.id == product.id,
    );

    if (existingIndex >= 0) {
      _items[existingIndex] = _items[existingIndex].copyWith(
        quantity: _items[existingIndex].quantity + 1,
      );
    } else {
      _items.add(CartItem(product: product, quantity: 1));
    }
    notifyListeners();
  }

  void removeFromCart(int productId) {
    _items.removeWhere((item) => item.product.id == productId);
    notifyListeners();
  }

  void updateQuantity(int productId, int quantity) {
    if (quantity <= 0) {
      removeFromCart(productId);
      return;
    }

    final index = _items.indexWhere((item) => item.product.id == productId);
    if (index >= 0) {
      _items[index] = _items[index].copyWith(quantity: quantity);
      notifyListeners();
    }
  }

  void toggleFavorite(Product product) {
    if (isFavorite(product.id)) {
      _favorites.removeWhere((p) => p.id == product.id);
    } else {
      _favorites.add(product);
    }
    notifyListeners();
  }

  void clearCart() {
    _items.clear();
    notifyListeners();
  }
}

class ProductsProvider extends ChangeNotifier {
  List<Product> _products = [];
  bool _isLoading = false;
  String? _error;
  String _searchQuery = '';
  String _selectedCategory = 'all';

  List<Product> get products {
    var filtered = _products;

    if (_selectedCategory != 'all') {
      // 简化的分类过滤逻辑
      filtered = filtered.where((p) => p.description.contains(_selectedCategory)).toList();
    }

    if (_searchQuery.isNotEmpty) {
      filtered = filtered.where((p) => 
        p.name.toLowerCase().contains(_searchQuery.toLowerCase()) ||
        p.description.toLowerCase().contains(_searchQuery.toLowerCase())
      ).toList();
    }

    return filtered;
  }

  bool get isLoading => _isLoading;
  String? get error => _error;
  String get searchQuery => _searchQuery;
  String get selectedCategory => _selectedCategory;

  Future<void> loadProducts() async {
    _isLoading = true;
    _error = null;
    notifyListeners();

    try {
      await Future.delayed(Duration(seconds: 2)); // 模拟网络延迟
      
      _products = [
        Product(id: 1, name: 'iPhone 14', price: 7999.0, imageUrl: 'https://example.com/iphone.jpg', description: '电子产品 手机'),
        Product(id: 2, name: 'MacBook Pro', price: 19999.0, imageUrl: 'https://example.com/macbook.jpg', description: '电子产品 电脑'),
        Product(id: 3, name: '咖啡', price: 25.0, imageUrl: 'https://example.com/coffee.jpg', description: '食品 饮料'),
        Product(id: 4, name: '书籍', price: 59.0, imageUrl: 'https://example.com/book.jpg', description: '图书 学习'),

    } catch (e) {
      _error = e.toString();
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  void setSearchQuery(String query) {
    _searchQuery = query;
    notifyListeners();
  }

  void setCategory(String category) {
    _selectedCategory = category;
    notifyListeners();
  }

  void clearFilters() {
    _searchQuery = '';
    _selectedCategory = 'all';
    notifyListeners();
  }
}

class CategoryFilter extends StatelessWidget {
  final List<String> categories = ['all', '电子产品', '食品', '图书', '服装'];

  @override
  Widget build(BuildContext context) {
    return Consumer<ProductsProvider>(
      builder: (context, productsProvider, child) {
        return Container(
          height: 50,
          child: ListView.builder(
            scrollDirection: Axis.horizontal,
            itemCount: categories.length,
            itemBuilder: (context, index) {

class ProductSearchDelegate extends SearchDelegate<String> {
  @override
  List<Widget> buildActions(BuildContext context) {
    return [
      IconButton(
        icon: Icon(Icons.clear),
        onPressed: () => query = '',
      ),
    ];
  }

  @override
  Widget buildLeading(BuildContext context) {
    return IconButton(
      icon: Icon(Icons.arrow_back),
      onPressed: () => close(context, ''),
    );
  }

  @override
  Widget buildResults(BuildContext context) {
    return _buildSearchResults(context);
  }

  @override
  Widget buildSuggestions(BuildContext context) {
    return _buildSearchResults(context);
  }

  Widget _buildSearchResults(BuildContext context) {
    if (query.isEmpty) {
      return Center(child: Text('请输入搜索关键词'));
    }

    return Consumer<ProductsProvider>(
      builder: (context, productsProvider, child) {
        final filteredProducts = productsProvider._products.where((product) =>
          product.name.toLowerCase().contains(query.toLowerCase()) ||
          product.description.toLowerCase().contains(query.toLowerCase())
        ).toList();

        return ListView.builder(
          itemCount: filteredProducts.length,
          itemBuilder: (context, index) {
            return ProductListTile(product: filteredProducts[index]);
          },
        );
      },
    );
  }
}
```

## 性能优化与最佳实践

### 1. 避免不必要的重建

```dart
// ❌ 不好的做法
class BadWidget extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final provider = Provider.of<MyProvider>(context); // 监听整个 Provider
    return Text('只需要显示: ${provider.someSpecificValue}');
  }
}

// ✅ 好的做法
class GoodWidget extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Selector<MyProvider, String>(
      selector: (context, provider) => provider.someSpecificValue,
      builder: (context, value, child) {
        return Text('只需要显示: $value');
      },
    );
  }
}
```

### 2. 正确使用 listen 参数

```dart
class EventHandlingExample extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return ElevatedButton(
      onPressed: () {
        // ✅ 事件处理中不监听变化
        context.read<CounterProvider>().increment();
        // 或者
        Provider.of<CounterProvider>(context, listen: false).increment();
      },
      child: Consumer<CounterProvider>(
        builder: (context, counter, child) {
          // ✅ UI 构建时监听变化
          return Text('计数: ${counter.count}');
        },
      ),
    );
  }
}
```

### 3. 合理拆分 Provider

```dart
// ❌ 单一大型 Provider

  // 用户相关

}

  // 只管理购物车相关状态

  // 只管理设置相关状态
```

## 总结

Provider 是 Flutter 生态中最成熟的状态管理方案之一：

1. **ChangeNotifier** - 简单易用的状态通知机制
2. **Consumer/Selector** - 精确控制组件重建
3. **多 Provider** - 职责分离，便于维护
4. **性能优化** - 合理使用 listen 参数和 Selector
5. **最佳实践** - 避免单一大型 Provider，保持状态的独立性

Provider 特别适合中小型应用的状态管理，学习成本低，社区支持好。下一章我们将学习更加强大的 Bloc/Cubit 状态管理方案。
