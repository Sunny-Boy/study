# 039. 完成新闻资讯应用

## 1. 项目概述与功能设计

### 1.1 项目目标
构建一个新闻资讯应用，实现以下功能：
- 新闻列表展示与分类浏览
- 新闻详情页面
- 下拉刷新与上拉加载更多（分页）
- 网络请求与错误处理
- 图片缓存优化

### 1.2 技术选型
- **网络请求**：dio（HTTP 客户端）
- **状态管理**：Provider
- **图片加载**：cached_network_image
- **下拉刷新**：pull_to_refresh
- **JSON 解析**：built_value 或手动解析

// models/news_article.dart
class NewsArticle {
  final String id;
  final String title;
  final String description;
  final String content;
  final String imageUrl;
  final String source;
  final String author;
  final DateTime publishedAt;
  final String category;
  final String url;

  NewsArticle({
    required this.id,
    required this.title,
    required this.description,
    required this.content,
    required this.imageUrl,
    required this.source,
    required this.author,
    required this.publishedAt,
    required this.category,
    required this.url,
  });

  factory NewsArticle.fromJson(Map<String, dynamic> json) {
    return NewsArticle(
      id: json['id']?.toString() ?? '',
      title: json['title'] ?? '无标题',
      description: json['description'] ?? '',
      content: json['content'] ?? '',
      imageUrl: json['urlToImage'] ?? json['image'] ?? '',
      source: json['source']?['name'] ?? json['source'] ?? '未知来源',
      author: json['author'] ?? '佚名',
      publishedAt: json['publishedAt'] != null
          ? DateTime.parse(json['publishedAt'])
          : DateTime.now(),
      category: json['category'] ?? 'general',
      url: json['url'] ?? '',
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'title': title,
      'description': description,
      'content': content,
      'urlToImage': imageUrl,
      'source': {'name': source},
      'author': author,
      'publishedAt': publishedAt.toIso8601String(),
      'category': category,
      'url': url,
    };
  }
}

### 3.2 网络服务类
```dart
// services/news_service.dart
import 'package:dio/dio.dart';
import '../config/api_config.dart';
import '../models/news_article.dart';

class NewsService {
  late final Dio _dio;

  NewsService() {
    _dio = Dio(BaseOptions(
      baseUrl: ApiConfig.baseUrl,
      connectTimeout: ApiConfig.timeout,
      receiveTimeout: ApiConfig.timeout,
    ));

    // 添加拦截器（日志、错误处理）
    _dio.interceptors.add(LogInterceptor(
      requestBody: true,
      responseBody: true,
    ));
  }

  // 获取新闻列表
  Future<List<NewsArticle>> fetchNews({
    required String category,
    int page = 1,
  }) async {
    try {
      final response = await _dio.get(
        '/top-headlines',
        queryParameters: {
          'apiKey': ApiConfig.apiKey,
          'category': category,
          'page': page,
          'pageSize': ApiConfig.pageSize,
          'country': 'us', // 可根据需要调整
        },
      );

      if (response.statusCode == 200) {
        final data = response.data;
        final articles = (data['articles'] as List)
            .map((json) => NewsArticle.fromJson(json))
            .toList();
        return articles;
      } else {
        throw DioException(
          requestOptions: response.requestOptions,
          response: response,
          type: DioExceptionType.badResponse,
        );
      }
    } on DioException catch (e) {
      throw _handleDioError(e);
    }
  }

  // 搜索新闻
  Future<List<NewsArticle>> searchNews({
    required String query,
    int page = 1,
  }) async {
    try {
      final response = await _dio.get(
        '/everything',
        queryParameters: {
          'apiKey': ApiConfig.apiKey,
          'q': query,
          'page': page,
          'pageSize': ApiConfig.pageSize,
          'sortBy': 'publishedAt',
        },
      );

      if (response.statusCode == 200) {
        final data = response.data;
        final articles = (data['articles'] as List)
            .map((json) => NewsArticle.fromJson(json))
            .toList();
        return articles;
      } else {
        throw Exception('搜索失败');
      }
    } on DioException catch (e) {
      throw _handleDioError(e);
    }
  }

  // 统一错误处理
  Exception _handleDioError(DioException error) {
    switch (error.type) {
      case DioExceptionType.connectionTimeout:
      case DioExceptionType.sendTimeout:
      case DioExceptionType.receiveTimeout:
        return Exception('网络连接超时，请检查网络设置');
      case DioExceptionType.badResponse:
        final statusCode = error.response?.statusCode;
        if (statusCode == 401) {
          return Exception('API 密钥无效');
        } else if (statusCode == 429) {
          return Exception('请求过于频繁，请稍后再试');
        }
        return Exception('服务器错误: $statusCode');
      case DioExceptionType.cancel:
        return Exception('请求已取消');
      default:
        return Exception('网络异常: ${error.message}');
    }
  }
}
```

## 4. 状态管理实现

### 4.1 NewsProvider
```dart
// providers/news_provider.dart
import 'package:flutter/foundation.dart';
import '../models/news_article.dart';
import '../models/news_category.dart';
import '../services/news_service.dart';

enum LoadingState { idle, loading, success, error }

class NewsProvider with ChangeNotifier {
  final NewsService _newsService = NewsService();

  // 当前分类
  NewsCategory _currentCategory = NewsCategory.general;
  NewsCategory get currentCategory => _currentCategory;

  // 新闻列表
  List<NewsArticle> _articles = [];
  List<NewsArticle> get articles => _articles;

  // 加载状态
  LoadingState _loadingState = LoadingState.idle;
  LoadingState get loadingState => _loadingState;

  // 错误信息
  String? _errorMessage;
  String? get errorMessage => _errorMessage;

  // 分页
  int _currentPage = 1;
  bool _hasMore = true;
  bool get hasMore => _hasMore;

  // 切换分类
  void changeCategory(NewsCategory category) {
    if (_currentCategory != category) {
      _currentCategory = category;
      _currentPage = 1;
      _articles.clear();
      notifyListeners();
      fetchNews();
    }
  }

  // 获取新闻（首次加载或刷新）
  Future<void> fetchNews({bool isRefresh = false}) async {
    if (isRefresh) {
      _currentPage = 1;
      _articles.clear();
      _hasMore = true;
    }

    if (_loadingState == LoadingState.loading) return;

    _loadingState = LoadingState.loading;
    _errorMessage = null;
    notifyListeners();

    try {
      final newArticles = await _newsService.fetchNews(
        category: _currentCategory.value,
        page: _currentPage,
      );

      if (isRefresh) {
        _articles = newArticles;
      } else {
        _articles.addAll(newArticles);
      }

      _hasMore = newArticles.length >= ApiConfig.pageSize;
      _loadingState = LoadingState.success;
    } catch (e) {
      _errorMessage = e.toString();
      _loadingState = LoadingState.error;
    }

    notifyListeners();
  }

  // 加载更多
  Future<void> loadMore() async {
    if (!_hasMore || _loadingState == LoadingState.loading) return;

    _currentPage++;
    await fetchNews();
  }

  // 搜索新闻
  Future<void> searchNews(String query) async {
    if (query.trim().isEmpty) {
      fetchNews(isRefresh: true);
      return;
    }

    _loadingState = LoadingState.loading;
    _errorMessage = null;
    notifyListeners();

    try {
      final results = await _newsService.searchNews(
        query: query,
        page: 1,
      );
      _articles = results;
      _loadingState = LoadingState.success;
    } catch (e) {
      _errorMessage = e.toString();
      _loadingState = LoadingState.error;
    }

    notifyListeners();
  }
}
```

// screens/news_list_screen.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import 'package:pull_to_refresh/pull_to_refresh.dart';
import '../models/news_category.dart';
import '../providers/news_provider.dart';
import '../widgets/news_card.dart';
import '../widgets/error_widget.dart';
import 'news_detail_screen.dart';

class NewsListScreen extends StatefulWidget {
  const NewsListScreen({Key? key}) : super(key: key);

  @override
  State<NewsListScreen> createState() => _NewsListScreenState();
}

class _NewsListScreenState extends State<NewsListScreen> {
  final RefreshController _refreshController = RefreshController();
  final TextEditingController _searchController = TextEditingController();

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) {
      Provider.of<NewsProvider>(context, listen: false).fetchNews();
    });
  }

  @override
  void dispose() {
    _refreshController.dispose();
    _searchController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('新闻资讯'),
        bottom: PreferredSize(
          preferredSize: const Size.fromHeight(100),
          child: Column(
            children: [
              _buildSearchBar(),
              _buildCategoryTabs(),
            ],
          ),
        ),
      ),
      body: _buildBody(),
    );
  }

  Widget _buildSearchBar() {
    return Padding(
      padding: const EdgeInsets.all(8.0),
      child: TextField(
        controller: _searchController,
        decoration: InputDecoration(
          hintText: '搜索新闻...',
          prefixIcon: const Icon(Icons.search),
          suffixIcon: IconButton(
            icon: const Icon(Icons.clear),
            onPressed: () {
              _searchController.clear();
              Provider.of<NewsProvider>(context, listen: false)
                  .fetchNews(isRefresh: true);
            },
          ),
          border: OutlineInputBorder(
            borderRadius: BorderRadius.circular(25),
          ),
          filled: true,
          fillColor: Colors.white,
        ),
        onSubmitted: (query) {
          Provider.of<NewsProvider>(context, listen: false).searchNews(query);
        },
      ),
    );
  }

  Widget _buildCategoryTabs() {
    return Consumer<NewsProvider>(
      builder: (context, provider, child) {
        return SizedBox(
          height: 40,
          child: ListView.builder(
            scrollDirection: Axis.horizontal,
            padding: const EdgeInsets.symmetric(horizontal: 8),
            itemCount: NewsCategory.values.length,
            itemBuilder: (context, index) {
              final category = NewsCategory.values[index];
              final isSelected = provider.currentCategory == category;

              return Padding(
                padding: const EdgeInsets.only(right: 8),
                child: ChoiceChip(
                  label: Text(category.label),
                  selected: isSelected,
                  onSelected: (selected) {
                    if (selected) {
                      provider.changeCategory(category);
                    }
                  },
                ),
              );
            },
          ),
        );
      },
    );
  }

  Widget _buildBody() {
    return Consumer<NewsProvider>(
      builder: (context, provider, child) {
        if (provider.loadingState == LoadingState.loading &&
            provider.articles.isEmpty) {
          return const Center(child: CircularProgressIndicator());
        }

        if (provider.loadingState == LoadingState.error &&
            provider.articles.isEmpty) {
          return ErrorDisplayWidget(
            message: provider.errorMessage ?? '加载失败',
            onRetry: () => provider.fetchNews(isRefresh: true),
          );
        }

        if (provider.articles.isEmpty) {
          return const Center(child: Text('暂无新闻'));
        }

        return SmartRefresher(
          controller: _refreshController,
          enablePullDown: true,
          enablePullUp: provider.hasMore,
          onRefresh: () async {
            await provider.fetchNews(isRefresh: true);
            _refreshController.refreshCompleted();
          },
          onLoading: () async {
            await provider.loadMore();
            _refreshController.loadComplete();
          },
          child: ListView.builder(
            itemCount: provider.articles.length,
            itemBuilder: (context, index) {
              final article = provider.articles[index];
              return NewsCard(
                article: article,
                onTap: () => _navigateToDetail(article),
              );
            },
          ),
        );
      },
    );
  }

  void _navigateToDetail(article) {
    Navigator.of(context).push(
      MaterialPageRoute(
        builder: (context) => NewsDetailScreen(article: article),
      ),
    );
  }
}

### 7.2 请求重试
```dart
Future<Response> _retryRequest(
  Future<Response> Function() request, {
  int maxRetries = 3,
}) async {
  int attempts = 0;
  while (attempts < maxRetries) {
    try {
      return await request();
    } catch (e) {
      attempts++;
      if (attempts >= maxRetries) rethrow;
      await Future.delayed(Duration(seconds: attempts));
    }
  }
  throw Exception('请求失败');
}
```
