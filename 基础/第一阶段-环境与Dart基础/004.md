class Student {

  // 3. 工厂构造函数
  factory Student.fromJson(Map<String, dynamic> json) {

    );
  }

  // 5. const构造函数
  const Student.anonymous() : 
    name = '匿名',
    age = 0,
    studentId = 'ANON',
    courses = const [];
  
  // 6. 初始化列表构造函数
  Student.withValidation(String inputName, int inputAge, String inputId)
    : assert(inputName.isNotEmpty, '姓名不能为空'),
      assert(inputAge >= 0, '年龄不能为负数'),
      name = inputName.trim(),
      age = inputAge,
      studentId = inputId.toUpperCase(),

    }
  }
  
  @override

  }

  }

    for (Animal animal in animals) {
      animal.makeSound();  // 多态调用，实际执行子类重写的方法
    }
  }
  
  void putAllAnimalsToSleep() {
    print('\n😴 所有动物开始休息:');
    for (Animal animal in animals) {
      animal.sleep();
    }
  }
  
  // 类型检查和转换
  void performSpecialActions() {
    print('\n🎯 执行特殊动作:');
    for (Animal animal in animals) {
      // 运行时类型检查
      if (animal is Dog) {
        Dog dog = animal; // 自动类型转换
        dog.wagTail();
        dog.guard();
      } else if (animal is Cat) {
        Cat cat = animal as Cat; // 显式类型转换
        cat.climb();
        cat.hunt();
      }
    }
  }
  
  void showAllAnimalsInfo() {
    print('\n📋 动物信息列表:');
    for (int i = 0; i < animals.length; i++) {
      print('${i + 1}. ${animals[i].getInfo()}');
    }
  }

// 抽象基类
abstract class Shape {
  String name;
  
  Shape(this.name);
  
  // 抽象方法（必须被子类实现）
  double calculateArea();
  double calculatePerimeter();
  
  // 具体方法（子类可直接使用）
  void displayInfo() {
    print('形状: $name');
    print('面积: ${calculateArea().toStringAsFixed(2)}');
    print('周长: ${calculatePerimeter().toStringAsFixed(2)}');
  }
  
  // 抽象getter
  String get shapeType;
}

// 实现抽象类 - 圆形
class Circle extends Shape {
  double radius;
  
  Circle(this.radius) : super('圆形');
  
  @override
  double calculateArea() => 3.14159 * radius * radius;
  
  @override
  double calculatePerimeter() => 2 * 3.14159 * radius;
  
  @override
  String get shapeType => '圆形';
}

// 实现抽象类 - 矩形
class Rectangle extends Shape {
  double width;
  double height;
  
  Rectangle(this.width, this.height) : super('矩形');
  
  @override
  double calculateArea() => width * height;
  
  @override
  double calculatePerimeter() => 2 * (width + height);
  
  @override
  String get shapeType => '矩形';
  
  // 矩形特有方法
  bool get isSquare => width == height;
}

// 定义接口（使用抽象类）
abstract class Drawable {
  void draw();
  void erase();
}

abstract class Movable {
  void moveTo(double x, double y);
  void moveBy(double deltaX, double deltaY);
}

abstract class Scalable {
  void scale(double factor);
}

// 多接口实现
class GraphicElement implements Drawable, Movable, Scalable {
  double x, y;
  double scaleX, scaleY;
  String elementType;
  
  GraphicElement(this.elementType, this.x, this.y) 
    : scaleX = 1.0, scaleY = 1.0;
  
  // 实现Drawable接口
  @override
  void draw() {
    print('🎨 绘制$elementType于位置($x, $y)，缩放比例($scaleX, $scaleY)');
  }
  
  @override
  void erase() {
    print('🗑️  擦除$elementType');
  }
  
  // 实现Movable接口
  @override
  void moveTo(double newX, double newY) {
    x = newX;
    y = newY;
    print('📍 $elementType移动到位置($x, $y)');
  }
  
  @override
  void moveBy(double deltaX, double deltaY) {
    x += deltaX;
    y += deltaY;
    print('↗️ $elementType移动了($deltaX, $deltaY)，当前位置($x, $y)');
  }
  
  // 实现Scalable接口
  @override
  void scale(double factor) {
    scaleX *= factor;
    scaleY *= factor;
    print('📏 $elementType缩放了${factor}倍，当前比例($scaleX, $scaleY)');
  }
}

// 接口使用示例
void demonstrateInterfaces() {
  GraphicElement circle = GraphicElement('圆形', 10, 20);
  GraphicElement rectangle = GraphicElement('矩形', 50, 100);
  
  List<GraphicElement> elements = [circle, rectangle];
  
  print('🎯 图形元素操作演示:');
  for (GraphicElement element in elements) {
    element.draw();
    element.moveTo(100, 200);
    element.scale(1.5);
    element.moveBy(10, -5);
    element.erase();
    print('-' * 30);
  }
}

class Vector2D {
  double x, y;
  
  Vector2D(this.x, this.y);
  
  // 重载 + 操作符
  Vector2D operator +(Vector2D other) {
    return Vector2D(x + other.x, y + other.y);
  }
  
  // 重载 - 操作符
  Vector2D operator -(Vector2D other) {
    return Vector2D(x - other.x, y - other.y);
  }
  
  // 重载 * 操作符（标量乘法）
  Vector2D operator *(double scalar) {
    return Vector2D(x * scalar, y * scalar);
  }
  
  // 重载 == 操作符
  @override
  bool operator ==(Object other) {
    if (other is Vector2D) {
      return x == other.x && y == other.y;
    }
    return false;
  }
  
  // 重载hashCode（与==一起使用）
  @override
  int get hashCode => x.hashCode ^ y.hashCode;
  
  // 重载 [] 操作符
  double operator [](int index) {
    switch (index) {
      case 0: return x;
      case 1: return y;
      default: throw ArgumentError('Index out of bounds');
    }
  }
  
  // 重载 []= 操作符
  void operator []=(int index, double value) {
    switch (index) {
      case 0: x = value; break;
      case 1: y = value; break;
      default: throw ArgumentError('Index out of bounds');
    }
  }
  
  // 计算向量长度
  double get length => math.sqrt(x * x + y * y);
  
  @override
  String toString() => 'Vector2D($x, $y)';
}

// 泛型栈实现
class Stack<T> {
  List<T> _items = [];
  
  // 入栈
  void push(T item) {
    _items.add(item);
    print('✅ 入栈: $item');
  }
  
  // 出栈
  T? pop() {
    if (_items.isEmpty) {
      print('⚠️ 栈为空，无法出栈');
      return null;
    }
    T item = _items.removeLast();
    print('📤 出栈: $item');
    return item;
  }
  
  // 查看栈顶元素
  T? peek() {
    if (_items.isEmpty) return null;
    return _items.last;
  }
  
  // 栈是否为空
  bool get isEmpty => _items.isEmpty;
  
  // 栈的大小
  int get size => _items.length;
  
  // 清空栈
  void clear() {
    _items.clear();
    print('🗑️ 栈已清空');
  }
  
  @override
  String toString() {
    return 'Stack(${_items.reversed.join(' <- ')})';
  }
}

// 泛型队列实现
class Queue<T> {
  List<T> _items = [];
  
  void enqueue(T item) {
    _items.add(item);
    print('➡️ 入队: $item');
  }
  
  T? dequeue() {
    if (_items.isEmpty) {
      print('⚠️ 队列为空');
      return null;
    }
    T item = _items.removeAt(0);
    print('⬅️ 出队: $item');
    return item;
  }
  
  T? get front => _items.isEmpty ? null : _items.first;
  bool get isEmpty => _items.isEmpty;
  int get length => _items.length;
  
  @override
  String toString() => 'Queue(${_items.join(' <- ')})';
}

// 泛型使用示例

## 大白话解释

面向对象编程就像是在**搭建乐高积木**：

- **类（Class）** = **积木的设计图纸**，定义了积木的形状和功能
- **对象（Object）** = **具体的积木块**，根据图纸制造出来的实物
- **继承（Inheritance）** = **积木的改良版本**，在原版基础上增加新功能
- **多态（Polymorphism）** = **同样的接口，不同的实现**，比如都是"车"，但有小轿车、货车、摩托车
- **抽象类（Abstract Class）** = **半成品图纸**，定义了基本结构，但需要完善细节
- **接口（Interface）** = **功能规范**，说明积木应该有哪些插槽和功能
- **Mixin** = **功能插件**，可以给积木添加额外的能力

**内存管理**就像是**自动回收垃圾的工厂**：
- 工厂会自动清理不再使用的废料（垃圾回收）
- 新产品先放在临时区域（New Space）
- 长期使用的产品移到仓库区（Old Space）
- 定期清理没人要的产品，释放空间

class _InteractiveCounterState extends State<InteractiveCounter> 
    with TickerProviderStateMixin {
  
  late int _currentValue;
  late AnimationController _animationController;
  late Animation<double> _scaleAnimation;
  
  @override
  void initState() {
    super.initState();
    _currentValue = widget.initialValue;
    
    // 初始化动画控制器
    _animationController = AnimationController(
      duration: Duration(milliseconds: 150),
      vsync: this,
    );
    
    _scaleAnimation = Tween<double>(
      begin: 1.0,
      end: 1.2,
    ).animate(CurvedAnimation(
      parent: _animationController,
      curve: Curves.elasticOut,
    ));
  }
  
  @override
  void dispose() {
    _animationController.dispose();
    super.dispose();
  }
  
  void _increment() {
    if (_currentValue < widget.maxValue) {
      setState(() {
        _currentValue++;
      });
      _triggerAnimation();
      widget.onValueChanged?.call(_currentValue);
    }
  }
  
  void _decrement() {
    if (_currentValue > widget.minValue) {
      setState(() {
        _currentValue--;
      });
      _triggerAnimation();
      widget.onValueChanged?.call(_currentValue);
    }
  }

    _animationController.forward().then((_) {
      _animationController.reverse();
    });
  }

// 基础实体类
abstract class Entity {
  final String id;
  final DateTime createdAt;
  DateTime updatedAt;
  
  Entity(this.id) : 
    createdAt = DateTime.now(),
    updatedAt = DateTime.now();
  
  void updateTimestamp() {
    updatedAt = DateTime.now();
  }
}

// 图书类
class Book extends Entity {
  String title;
  String author;
  String isbn;
  double price;
  int stock;
  List<String> categories;
  
  Book({
    required String id,
    required this.title,
    required this.author,
    required this.isbn,
    required this.price,
    this.stock = 0,
    List<String>? categories,
  }) : categories = categories ?? [], super(id);
  
  // 检查是否有库存
  bool get isAvailable => stock > 0;
  
  // 借出图书
  bool borrow() {
    if (isAvailable) {
      stock--;
      updateTimestamp();
      return true;
    }
    return false;
  }
  
  // 归还图书
  void returnBook() {
    stock++;
    updateTimestamp();
  }
  
  @override
  String toString() {
    return 'Book(title: $title, author: $author, stock: $stock)';
  }
}

// 借阅记录类
class BorrowRecord extends Entity {
  final String readerId;
  final String bookId;
  final DateTime borrowDate;
  DateTime? returnDate;
  
  BorrowRecord({
    required String id,
    required this.readerId,
    required this.bookId,
  }) : borrowDate = DateTime.now(), super(id);
  
  // 是否已归还
  bool get isReturned => returnDate != null;
  
  // 是否逾期（假设借阅期限30天）
  bool get isOverdue {
    if (isReturned) return false;
    return DateTime.now().difference(borrowDate).inDays > 30;
  }
  
  // 归还图书
  void markReturned() {
    returnDate = DateTime.now();
    updateTimestamp();
  }
  
  @override
  String toString() {
    String status = isReturned ? '已归还' : (isOverdue ? '逾期' : '借阅中');
    return 'BorrowRecord(reader: $readerId, book: $bookId, status: $status)';
  }
}

// 图书管理系统
class LibraryManager {
  Map<String, Book> _books = {};
  Map<String, User> _users = {};
  Map<String, BorrowRecord> _borrowRecords = {};
  
  // 添加图书
  void addBook(Book book) {
    _books[book.id] = book;
    print('✅ 添加图书成功: ${book.title}');
  }
  
  // 注册用户
  void registerUser(User user) {
    _users[user.id] = user;
    print('✅ 用户注册成功: ${user.name} (${user.userType})');
  }
  
  // 借书流程
  bool borrowBook(String readerId, String bookId) {
    Reader? reader = _users[readerId] as Reader?;
    Book? book = _books[bookId];
    
    if (reader == null) {
      print('❌ 读者不存在: $readerId');
      return false;
    }
    
    if (book == null) {
      print('❌ 图书不存在: $bookId');
      return false;
    }
    
    if (!reader.canBorrow()) {
      print('❌ ${reader.name} 已达到借阅上限');
      return false;
    }
    
    if (!book.isAvailable) {
      print('❌ 图书《${book.title}》暂无库存');
      return false;
    }
    
    // 执行借书操作
    if (book.borrow() && reader.borrowBook(bookId)) {
      String recordId = 'BR_${DateTime.now().millisecondsSinceEpoch}';
      BorrowRecord record = BorrowRecord(
        id: recordId,
        readerId: readerId,
        bookId: bookId,
      );
      _borrowRecords[recordId] = record;
      
      print('✅ ${reader.name} 借阅《${book.title}》成功');
      return true;
    }
    
    return false;
  }
  
  // 还书流程
  bool returnBook(String readerId, String bookId) {
    Reader? reader = _users[readerId] as Reader?;
    Book? book = _books[bookId];
    
    if (reader == null || book == null) {
      print('❌ 读者或图书不存在');
      return false;
    }
    
    // 查找借阅记录
    BorrowRecord? record = _borrowRecords.values.firstWhere(
      (r) => r.readerId == readerId && r.bookId == bookId && !r.isReturned,
      orElse: () => null as BorrowRecord,
    );
    
    if (record == null) {
      print('❌ 未找到借阅记录');
      return false;
    }
    
    // 执行还书操作
    book.returnBook();
    reader.returnBook(bookId);
    record.markReturned();
    
    String status = record.isOverdue ? '（逾期归还）' : '';
    print('✅ ${reader.name} 归还《${book.title}》成功 $status');
    
    return true;
  }
  
  // 显示系统统计
  void showStatistics() {
    print('\n📊 图书馆系统统计:');
    print('=' * 40);
    print('📚 图书总数: ${_books.length}');
    print('👥 用户总数: ${_users.length}');
    print('📋 借阅记录: ${_borrowRecords.length}');
    
    int availableBooks = _books.values.where((b) => b.isAvailable).length;
    print('📖 可借图书: $availableBooks');
    
    int activeRecords = _borrowRecords.values.where((r) => !r.isReturned).length;
    print('📊 当前借阅: $activeRecords');
    
    int overdueRecords = _borrowRecords.values.where((r) => r.isOverdue).length;
    print('⚠️  逾期记录: $overdueRecords');
  }
  
  // 显示所有图书

    if (_books.isEmpty) {
      print('暂无图书');
      return;
    }
    
    _books.values.forEach((book) {
      String status = book.isAvailable ? '✅ 可借' : '❌ 无库存';
      print('${book.title} - ${book.author} [$status]');
    });
  }
  
  // 显示所有用户
  void showAllUsers() {
    print('\n👥 用户列表:');
    if (_users.isEmpty) {
      print('暂无用户');
      return;
    }
    
    _users.values.forEach((user) {
      print('${user.name} (${user.userType}) - ${user.email}');
    });
  }
}
