### 自定义手势识别器

```dart
class CustomGestureRecognizer extends StatefulWidget {
  @override
  _CustomGestureRecognizerState createState() => _CustomGestureRecognizerState();
}

class _CustomGestureRecognizerState extends State<CustomGestureRecognizer> {
  List<Offset> _drawingPath = [];
  String _gesturePattern = '';

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('自定义手势识别'),
        actions: [
          IconButton(
            icon: Icon(Icons.clear),
            onPressed: _clearDrawing,
          ),
        ],
      ),
      body: Column(
        children: [
          Container(
            width: double.infinity,
            padding: EdgeInsets.all(16),
            color: Colors.grey.shade100,
            child: Text(
              _gesturePattern.isEmpty ? '在下方区域绘制手势' : '识别模式: $_gesturePattern',
              textAlign: TextAlign.center,
              style: TextStyle(fontSize: 16),
            ),
          ),
          
          Expanded(
            child: Container(
              width: double.infinity,
              color: Colors.black12,
              child: GestureDetector(
                onPanStart: _onPanStart,
                onPanUpdate: _onPanUpdate,
                onPanEnd: _onPanEnd,
                child: CustomPaint(
                  painter: DrawingPainter(_drawingPath),
                  size: Size.infinite,
                ),
              ),
            ),
          ),
          
          Container(
            padding: EdgeInsets.all(16),
            child: Column(
              children: [
                Text(
                  '手势提示：',
                  style: TextStyle(fontWeight: FontWeight.bold),
                ),
                SizedBox(height: 8),
                Wrap(
                  spacing: 8,
                  children: [
                    Chip(label: Text('圆形: 画圆')),
                    Chip(label: Text('直线: 画直线')),
                    Chip(label: Text('波浪: 来回画')),
                  ],
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }

  void _onPanStart(DragStartDetails details) {
    setState(() {
      _drawingPath.clear();
      _drawingPath.add(details.localPosition);
    });
  }

  void _onPanUpdate(DragUpdateDetails details) {
    setState(() {
      _drawingPath.add(details.localPosition);
    });
  }

  void _onPanEnd(DragEndDetails details) {
    _analyzeGesture();
  }

  void _analyzeGesture() {
    if (_drawingPath.length < 5) return;

    // 简单的手势识别算法
    final startPoint = _drawingPath.first;
    final endPoint = _drawingPath.last;
    final distance = (endPoint - startPoint).distance;

    String pattern = '';

    if (distance < 50) {
      // 起点和终点很近，可能是圆形或封闭图形
      pattern = '圆形/封闭图形';
    } else {
      // 计算路径的弯曲程度
      double totalDistance = 0;
      for (int i = 1; i < _drawingPath.length; i++) {
        totalDistance += (_drawingPath[i] - _drawingPath[i - 1]).distance;
      }

      final straightDistance = distance;
      final curvatureRatio = totalDistance / straightDistance;

      if (curvatureRatio < 1.2) {
        pattern = '直线';
      } else if (curvatureRatio > 2.0) {
        pattern = '波浪线/曲线';
      } else {
        pattern = '弯曲线';
      }
    }

    setState(() {
      _gesturePattern = pattern;
    });

    // 2秒后清除
    Future.delayed(Duration(seconds: 2), () {
      if (mounted) {
        _clearDrawing();
      }
    });
  }

  void _clearDrawing() {
    setState(() {
      _drawingPath.clear();
      _gesturePattern = '';
    });
  }
}

class DrawingPainter extends CustomPainter {
  final List<Offset> path;

  DrawingPainter(this.path);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.blue
      ..strokeWidth = 3.0
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round;

    if (path.length > 1) {
      final pathToDraw = Path();
      pathToDraw.moveTo(path.first.dx, path.first.dy);
      
      for (int i = 1; i < path.length; i++) {
        pathToDraw.lineTo(path[i].dx, path[i].dy);
      }
      
      canvas.drawPath(pathToDraw, paint);
    }

    // 绘制起点和终点
    if (path.isNotEmpty) {
      final pointPaint = Paint()
        ..color = Colors.red
        ..style = PaintingStyle.fill;
      
      canvas.drawCircle(path.first, 6, pointPaint);
      
      if (path.length > 1) {
        pointPaint.color = Colors.green;
        canvas.drawCircle(path.last, 6, pointPaint);
      }
    }
  }

  @override
  bool shouldRepaint(CustomPainter oldDelegate) => true;
}
```

```dart
                onPressed: _isProgressing ? null : _handlePress,
                style: ElevatedButton.styleFrom(
                  backgroundColor: Colors.transparent,
                  shadowColor: Colors.transparent,
                ),
                child: widget.child,
              ),
            ],
          );
        },
      ),
    );
  }

  Future<void> _handlePress() async {
    setState(() {
      _isProgressing = true;
    });
    
    _controller.forward();
    
    await Future.delayed(Duration(seconds: 3));
    
    if (mounted) {
      setState(() {
        _isProgressing = false;
      });
      _controller.reset();
      
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('进度操作完成')),
      );
    }
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }
}

          style: TextStyle(color: Colors.grey[600]),
          textAlign: TextAlign.center,
        ),
      ],
    );
  }

  void _addRipple(Offset position) {
    HapticFeedback.selectionClick();
    
    setState(() {
      _ripples.add(position);
    });
    
    _bounceController.forward().then((_) {
      _bounceController.reverse();
    });
    
    // 2秒后移除水波纹
    Future.delayed(Duration(seconds: 2), () {
      if (mounted && _ripples.isNotEmpty) {
        setState(() {
          _ripples.removeAt(0);
        });
      }
    });
  }

  @override
  void dispose() {
    _bounceController.dispose();
    super.dispose();
  }
}

class RipplePainter extends CustomPainter {
  final List<Offset> ripples;

  RipplePainter(this.ripples);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.blue.withOpacity(0.3)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2;

    for (int i = 0; i < ripples.length; i++) {
      final ripple = ripples[i];
      final progress = (DateTime.now().millisecondsSinceEpoch % 2000) / 2000.0;
      final radius = progress * 60;
      
      paint.color = Colors.blue.withOpacity(0.5 - progress * 0.5);
      canvas.drawCircle(ripple, radius, paint);
    }
  }

  @override
  bool shouldRepaint(CustomPainter oldDelegate) => true;
}
```

## 总结

用户交互处理是 Flutter 应用开发的核心技能：

1. **基础组件** - 熟练使用各种按钮、开关、选择组件
2. **手势识别** - 掌握 GestureDetector 和自定义手势
3. **滚动交互** - 实现流畅的列表和页面滚动效果
4. **反馈机制** - 提供及时、恰当的视觉和触觉反馈
5. **性能优化** - 保证交互的流畅性和响应速度
6. **设计原则** - 遵循用户体验设计的最佳实践

良好的交互设计能够显著提升用户体验，是区分优秀应用和普通应用的重要因素。在开发过程中要始终站在用户角度思考，提供直观、流畅、愉悦的交互体验。
