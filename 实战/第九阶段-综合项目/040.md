# 040. 搭建电商应用原型

## 1. 项目概述与架构

### 1.1 项目目标
构建一个电商应用原型，实现核心功能：
- 商品列表展示与分类筛选
- 商品详情页面（图片轮播、规格选择）
- 购物车管理（增删改、数量调整）
- 订单结算流程
- 使用 Bloc 进行复杂状态管理
- 本地数据缓存与用户偏好存储

### 1.2 技术架构
- **状态管理**：flutter_bloc（Bloc/Cubit）
- **网络请求**：dio + retrofit
- **本地存储**：hive（NoSQL 数据库）
- **路由管理**：go_router
- **依赖注入**：get_it

```dart
// data/models/product.dart
import 'package:hive/hive.dart';

part 'product.g.dart'; // 用于 Hive 代码生成

@HiveType(typeId: 0)
class Product {
  @HiveField(0)
  final String id;

  @HiveField(1)
  final String name;

  @HiveField(2)
  final String description;

  @HiveField(3)
  final double price;

  @HiveField(4)
  final double originalPrice;

  @HiveField(5)
  final List<String> images;

  @HiveField(6)
  final String category;

  @HiveField(7)
  final int stock;

  @HiveField(8)
  final double rating;

  @HiveField(9)
  final int reviewCount;

  @HiveField(10)
  final List<String> tags;

  Product({
    required this.id,
    required this.name,
    required this.description,
    required this.price,
    required this.originalPrice,
    required this.images,
    required this.category,
    required this.stock,
    this.rating = 0.0,
    this.reviewCount = 0,
    this.tags = const [],
  });

  factory Product.fromJson(Map<String, dynamic> json) {
    return Product(
      id: json['id']?.toString() ?? '',
      name: json['name'] ?? '',
      description: json['description'] ?? '',
      price: (json['price'] ?? 0).toDouble(),
      originalPrice: (json['originalPrice'] ?? json['price']).toDouble(),
      images: List<String>.from(json['images'] ?? []),
      category: json['category'] ?? '',
      stock: json['stock'] ?? 0,
      rating: (json['rating'] ?? 0.0).toDouble(),
      reviewCount: json['reviewCount'] ?? 0,
      tags: List<String>.from(json['tags'] ?? []),
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'name': name,
      'description': description,
      'price': price,
      'originalPrice': originalPrice,
      'images': images,
      'category': category,
      'stock': stock,
      'rating': rating,
      'reviewCount': reviewCount,
      'tags': tags,
    };
  }

  double get discountPercentage {
    if (originalPrice <= 0) return 0;
    return ((originalPrice - price) / originalPrice * 100).roundToDouble();
  }

  bool get isInStock => stock > 0;
}
```

```dart
// data/repositories/product_repository.dart
import '../models/product.dart';
import '../datasources/remote/api_service.dart';
import '../datasources/local/hive_service.dart';

class ProductRepository {
  final ApiService _apiService;
  final HiveService _hiveService;

  ProductRepository(this._apiService, this._hiveService);

  // 获取商品列表（优先使用缓存）
  Future<List<Product>> getProducts({String? category}) async {
    try {
      // 尝试从网络获取
      final products = await _apiService.fetchProducts(category: category);
      
      // 缓存到本地
      await _hiveService.saveProducts(products);
      
      return products;
    } catch (e) {
      // 网络失败时返回缓存数据
      return _hiveService.getCachedProducts(category: category);
    }
  }

  // 获取商品详情
  Future<Product?> getProductById(String id) async {
    try {
      return await _apiService.fetchProductById(id);
    } catch (e) {
      return _hiveService.getCachedProduct(id);
    }
  }

  // 搜索商品
  Future<List<Product>> searchProducts(String query) async {
    try {
      return await _apiService.searchProducts(query);
    } catch (e) {
      return _hiveService.searchCachedProducts(query);
    }
  }
}
```

### 3.2 购物车仓库
```dart
// data/repositories/cart_repository.dart
import '../models/cart_item.dart';
import '../models/product.dart';
import '../datasources/local/hive_service.dart';

class CartRepository {
  final HiveService _hiveService;

  CartRepository(this._hiveService);

  // 获取购物车
  Future<List<CartItem>> getCartItems() async {
    return await _hiveService.getCartItems();
  }

  // 添加到购物车
  Future<void> addToCart(Product product, {int quantity = 1}) async {
    final items = await getCartItems();
    final existingIndex = items.indexWhere((item) => item.product.id == product.id);

    if (existingIndex != -1) {
      // 商品已存在，增加数量
      items[existingIndex] = items[existingIndex].copyWith(
        quantity: items[existingIndex].quantity + quantity,
      );
    } else {
      // 新商品
      items.add(CartItem(product: product, quantity: quantity));
    }

    await _hiveService.saveCartItems(items);
  }

  // 更新数量
  Future<void> updateQuantity(String productId, int quantity) async {
    final items = await getCartItems();
    final index = items.indexWhere((item) => item.product.id == productId);

    if (index != -1) {
      if (quantity <= 0) {
        items.removeAt(index);
      } else {
        items[index] = items[index].copyWith(quantity: quantity);
      }
      await _hiveService.saveCartItems(items);
    }
  }

  // 移除商品
  Future<void> removeFromCart(String productId) async {
    final items = await getCartItems();
    items.removeWhere((item) => item.product.id == productId);
    await _hiveService.saveCartItems(items);
  }

  // 清空购物车
  Future<void> clearCart() async {
    await _hiveService.clearCart();
  }

  // 获取总金额
  Future<double> getTotalAmount() async {
    final items = await getCartItems();
    return items.fold(0, (sum, item) => sum + item.subtotal);
  }
}
```

```dart
// business_logic/blocs/product/product_bloc.dart
import 'package:flutter_bloc/flutter_bloc.dart';
import '../../../data/repositories/product_repository.dart';
import 'product_event.dart';
import 'product_state.dart';

class ProductBloc extends Bloc<ProductEvent, ProductState> {
  final ProductRepository _repository;

  ProductBloc(this._repository) : super(ProductInitial()) {
    on<LoadProducts>(_onLoadProducts);
    on<RefreshProducts>(_onRefreshProducts);
    on<SearchProducts>(_onSearchProducts);
  }

  Future<void> _onLoadProducts(
    LoadProducts event,
    Emitter<ProductState> emit,
  ) async {
    emit(ProductLoading());
    try {
      final products = await _repository.getProducts(
        category: event.category,
      );
      emit(ProductLoaded(products));
    } catch (e) {
      emit(ProductError(e.toString()));
    }
  }

  Future<void> _onRefreshProducts(
    RefreshProducts event,
    Emitter<ProductState> emit,
  ) async {
    try {
      final products = await _repository.getProducts();
      emit(ProductLoaded(products));
    } catch (e) {
      emit(ProductError(e.toString()));
    }
  }

  Future<void> _onSearchProducts(
    SearchProducts event,
    Emitter<ProductState> emit,
  ) async {
    emit(ProductLoading());
    try {
      final products = await _repository.searchProducts(event.query);
      emit(ProductLoaded(products));

      emit(ProductError(e.toString()));
```

```dart
class CartInitial extends CartState {}

class CartLoading extends CartState {}

class CartLoaded extends CartState {

  final double totalAmount;

  CartLoaded(this.items, this.totalAmount);

import 'package:flutter_bloc/flutter_bloc.dart';
import '../../../data/repositories/cart_repository.dart';
import 'cart_event.dart';
import 'cart_state.dart';

class CartBloc extends Bloc<CartEvent, CartState> {
  final CartRepository _repository;

  CartBloc(this._repository) : super(CartInitial()) {
    on<LoadCart>(_onLoadCart);
    on<AddToCart>(_onAddToCart);
    on<UpdateCartItem>(_onUpdateCartItem);
    on<RemoveFromCart>(_onRemoveFromCart);
    on<ClearCart>(_onClearCart);
  }

  Future<void> _onLoadCart(
    LoadCart event,
    Emitter<CartState> emit,
  ) async {
    emit(CartLoading());
    try {
      final items = await _repository.getCartItems();
      final total = await _repository.getTotalAmount();
      emit(CartLoaded(items, total));
    } catch (e) {
      emit(CartError(e.toString()));
    }
  }

  Future<void> _onAddToCart(
    AddToCart event,
    Emitter<CartState> emit,
  ) async {
    try {
      await _repository.addToCart(event.product, quantity: event.quantity);
      final items = await _repository.getCartItems();
      final total = await _repository.getTotalAmount();
      emit(CartLoaded(items, total));
    } catch (e) {
      emit(CartError(e.toString()));
    }
  }

  Future<void> _onUpdateCartItem(
    UpdateCartItem event,
    Emitter<CartState> emit,
  ) async {
    try {
      await _repository.updateQuantity(event.productId, event.quantity);
      final items = await _repository.getCartItems();
      final total = await _repository.getTotalAmount();
      emit(CartLoaded(items, total));
    } catch (e) {
      emit(CartError(e.toString()));
    }
  }

  Future<void> _onRemoveFromCart(
    RemoveFromCart event,
    Emitter<CartState> emit,
  ) async {
    try {
      await _repository.removeFromCart(event.productId);
      final items = await _repository.getCartItems();
      final total = await _repository.getTotalAmount();
      emit(CartLoaded(items, total));
    } catch (e) {
      emit(CartError(e.toString()));
    }
  }

  Future<void> _onClearCart(
    ClearCart event,
    Emitter<CartState> emit,
  ) async {
    try {
      await _repository.clearCart();
      emit(CartLoaded([], 0));
    } catch (e) {
      emit(CartError(e.toString()));
    }
  }
```

### 5.1 Hive 服务
```dart
// data/datasources/local/hive_service.dart
import 'package:hive_flutter/hive_flutter.dart';
import '../../models/product.dart';
import '../../models/cart_item.dart';

class HiveService {
  static const String _productsBox = 'products';
  static const String _cartBox = 'cart';

  Future<void> init() async {
    await Hive.initFlutter();
    
    // 注册适配器
    Hive.registerAdapter(ProductAdapter());
    Hive.registerAdapter(CartItemAdapter());

    // 打开盒子
    await Hive.openBox<Product>(_productsBox);
    await Hive.openBox<CartItem>(_cartBox);
  }

  // 商品缓存
  Future<void> saveProducts(List<Product> products) async {
    final box = Hive.box<Product>(_productsBox);
    await box.clear();
    for (var product in products) {
      await box.put(product.id, product);
    }
  }

  List<Product> getCachedProducts({String? category}) {
    final box = Hive.box<Product>(_productsBox);
    final allProducts = box.values.toList();
    
    if (category != null) {
      return allProducts.where((p) => p.category == category).toList();
    }
    return allProducts;
  }

  Product? getCachedProduct(String id) {
    final box = Hive.box<Product>(_productsBox);
    return box.get(id);
  }

  List<Product> searchCachedProducts(String query) {
    final box = Hive.box<Product>(_productsBox);
    return box.values
        .where((p) => p.name.toLowerCase().contains(query.toLowerCase()))
        .toList();
  }

  // 购物车管理
  Future<List<CartItem>> getCartItems() async {
    final box = Hive.box<CartItem>(_cartBox);
    return box.values.toList();
  }

  Future<void> saveCartItems(List<CartItem> items) async {
    final box = Hive.box<CartItem>(_cartBox);
    await box.clear();
    for (var i = 0; i < items.length; i++) {
      await box.put(i, items[i]);
    }
  }

  Future<void> clearCart() async {
    final box = Hive.box<CartItem>(_cartBox);
    await box.clear();
  }
}
```

```dart
class CartScreen extends StatelessWidget {

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('购物车'),
        actions: [
          BlocBuilder<CartBloc, CartState>(
            builder: (context, state) {
              if (state is CartLoaded && state.items.isNotEmpty) {

                  onPressed: () {
                    context.read<CartBloc>().add(ClearCart());
                  },
                  child: const Text('清空'),

              return const SizedBox.shrink();
            },
          ),
        ],
      ),
      body: BlocBuilder<CartBloc, CartState>(
        builder: (context, state) {
          if (state is CartLoading) {
            return const Center(child: CircularProgressIndicator());
          }

          if (state is CartError) {
            return Center(child: Text('错误: ${state.message}'));
          }

          if (state is CartLoaded) {

                child: Column(

                    Icon(
                      Icons.shopping_cart_outlined,

                      color: Colors.grey.shade400,

                  ],
                ),
              );

              children: [
                Expanded(
                  child: ListView.builder(
                    itemCount: state.items.length,
                    itemBuilder: (context, index) {
                      return CartItemWidget(
                        cartItem: state.items[index],
                        onQuantityChanged: (quantity) {
                          context.read<CartBloc>().add(UpdateCartItem(
                                state.items[index].product.id,
                                quantity,
                              ));

                                state.items[index].product.id,
                              ));
                        },
                      );

                ),
                _buildBottomBar(context, state.totalAmount),
              ],
            );
          }

          return const SizedBox.shrink();
        },
      ),
```

```dart
// core/di/injection.dart
import 'package:get_it/get_it.dart';
import '../../data/datasources/local/hive_service.dart';
import '../../data/datasources/remote/api_service.dart';
import '../../data/repositories/product_repository.dart';
import '../../data/repositories/cart_repository.dart';
import '../../business_logic/blocs/product/product_bloc.dart';
import '../../business_logic/blocs/cart/cart_bloc.dart';

final getIt = GetIt.instance;

Future<void> setupDependencies() async {
  // 服务层
  final hiveService = HiveService();
  await hiveService.init();
  getIt.registerSingleton<HiveService>(hiveService);
  
  getIt.registerLazySingleton<ApiService>(() => ApiService());

  // 仓库层
  getIt.registerLazySingleton<ProductRepository>(
    () => ProductRepository(getIt<ApiService>(), getIt<HiveService>()),
  );
  
  getIt.registerLazySingleton<CartRepository>(
    () => CartRepository(getIt<HiveService>()),
  );

  // Bloc/Cubit
  getIt.registerFactory<ProductBloc>(
    () => ProductBloc(getIt<ProductRepository>()),
  );
  
  getIt.registerFactory<CartBloc>(
    () => CartBloc(getIt<CartRepository>()),
  );
}
```

## 9. 总结与要点

### 9.1 核心技术
- **Bloc 模式**：事件驱动的状态管理
- **仓库模式**：数据层抽象与缓存策略
- **依赖注入**：解耦与可测试性
- **本地存储**：Hive NoSQL 数据库

- 清晰的分层架构（数据层、业务层、展示层）
- 统一的错误处理与状态管理
- 本地缓存优先策略（离线支持）
