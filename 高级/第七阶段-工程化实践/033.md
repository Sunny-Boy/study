// 相机服务封装
import 'package:camera/camera.dart';
import 'package:path_provider/path_provider.dart';
import 'dart:io';

class AdvancedCameraService {
  CameraController? _controller;
  List<CameraDescription> _cameras = [];
  
  // 初始化相机
  Future<bool> initializeCamera({
    CameraLensDirection direction = CameraLensDirection.back,
    ResolutionPreset resolution = ResolutionPreset.high,
  }) async {
    try {
      _cameras = await availableCameras();
      
      final camera = _cameras.firstWhere(
        (camera) => camera.lensDirection == direction,
        orElse: () => _cameras.first,
      );
      
      _controller = CameraController(
        camera,
        resolution,
        enableAudio: true,
        imageFormatGroup: ImageFormatGroup.jpeg,
      );
      
      await _controller!.initialize();
      return true;
    } catch (e) {
      print('相机初始化失败: $e');
      return false;
    }
  }
  
  // 拍照并保存
  Future<String?> takePicture() async {
    if (_controller == null || !_controller!.value.isInitialized) {
      return null;
    }
    
    try {
      final directory = await getApplicationDocumentsDirectory();
      final imagePath = '${directory.path}/image_${DateTime.now().millisecondsSinceEpoch}.jpg';
      
      final XFile picture = await _controller!.takePicture();
      await picture.saveTo(imagePath);
      
      return imagePath;
    } catch (e) {
      print('拍照失败: $e');
      return null;
    }
  }
  
  // 录制视频
  Future<String?> startVideoRecording() async {
    if (_controller == null || !_controller!.value.isInitialized) {
      return null;
    }
    
    try {
      await _controller!.startVideoRecording();
      return '录制开始';
    } catch (e) {
      print('开始录制失败: $e');
      return null;
    }
  }
  
  Future<String?> stopVideoRecording() async {
    if (_controller == null || !_controller!.value.isRecordingVideo) {
      return null;
    }
    
    try {
      final XFile videoFile = await _controller!.stopVideoRecording();
      
      final directory = await getApplicationDocumentsDirectory();
      final videoPath = '${directory.path}/video_${DateTime.now().millisecondsSinceEpoch}.mp4';
      
      await videoFile.saveTo(videoPath);
      return videoPath;
    } catch (e) {
      print('停止录制失败: $e');
      return null;
    }
  }
  
  // 切换摄像头
  Future<void> switchCamera() async {
    if (_cameras.length < 2) return;
    
    final currentDirection = _controller?.description.lensDirection;
    final newDirection = currentDirection == CameraLensDirection.back
        ? CameraLensDirection.front
        : CameraLensDirection.back;
    
    await _controller?.dispose();
    await initializeCamera(direction: newDirection);
  }
  
  // 设置闪光灯模式
  Future<void> setFlashMode(FlashMode mode) async {
    await _controller?.setFlashMode(mode);
  }
  
  // 设置对焦点
  Future<void> setFocusPoint(Offset point) async {
    await _controller?.setFocusPoint(point);
  }
  
  // 设置曝光点
  Future<void> setExposurePoint(Offset point) async {
    await _controller?.setExposurePoint(point);
  }
  
  // 设置缩放
  Future<void> setZoomLevel(double zoom) async {
    final minZoom = await _controller!.getMinZoomLevel();
    final maxZoom = await _controller!.getMaxZoomLevel();
    final clampedZoom = zoom.clamp(minZoom, maxZoom);
    await _controller?.setZoomLevel(clampedZoom);
  }
  
  // 释放资源
  void dispose() {
    _controller?.dispose();
  }
}

import 'package:geolocator/geolocator.dart';
import 'package:geocoding/geocoding.dart';

class LocationService {
  static const LocationSettings _locationSettings = LocationSettings(
    accuracy: LocationAccuracy.high,
    distanceFilter: 10,
  );
  
  // 检查定位权限
  static Future<bool> checkLocationPermission() async {
    bool serviceEnabled = await Geolocator.isLocationServiceEnabled();
    if (!serviceEnabled) {
      return false;
    }
    
    LocationPermission permission = await Geolocator.checkPermission();
    if (permission == LocationPermission.denied) {
      permission = await Geolocator.requestPermission();
      if (permission == LocationPermission.denied) {
        return false;
      }
    }
    
    if (permission == LocationPermission.deniedForever) {
      return false;
    }
    
    return true;
  }
  
  // 获取当前位置
  static Future<Position?> getCurrentLocation() async {
    try {
      if (!await checkLocationPermission()) {
        return null;
      }
      
      return await Geolocator.getCurrentPosition(
        desiredAccuracy: LocationAccuracy.high,
      );
    } catch (e) {
      print('获取位置失败: $e');
      return null;
    }
  }
  
  // 监听位置变化
  static Stream<Position> getPositionStream() {
    return Geolocator.getPositionStream(locationSettings: _locationSettings);
  }
  
  // 计算两点间距离
  static double calculateDistance(
    double lat1, double lon1,
    double lat2, double lon2,
  ) {
    return Geolocator.distanceBetween(lat1, lon1, lat2, lon2);
  }
  
  // 地理编码：地址转坐标
  static Future<List<Location>> getLocationFromAddress(String address) async {
    try {
      return await locationFromAddress(address);
    } catch (e) {
      print('地理编码失败: $e');
      return [];
    }
  }
  
  // 反向地理编码：坐标转地址
  static Future<List<Placemark>> getAddressFromLocation(
    double latitude, double longitude) async {
    try {
      return await placemarkFromCoordinates(latitude, longitude);
    } catch (e) {
      print('反向地理编码失败: $e');
      return [];
    }
  }
  
  // 格式化地址
  static String formatAddress(Placemark placemark) {
    final components = [
      placemark.country,
      placemark.administrativeArea,
      placemark.locality,
      placemark.thoroughfare,
      placemark.name,
    ];
    
    return components.where((c) => c != null && c.isNotEmpty).join(', ');
  }
}

// 定位管理器
class LocationManager {
  StreamSubscription<Position>? _positionSubscription;
  Position? _currentPosition;
  List<Position> _locationHistory = [];
  
  // 开始跟踪位置
  void startLocationTracking({
    required Function(Position) onLocationUpdate,
    Function(String)? onError,
  }) {
    _positionSubscription = LocationService.getPositionStream().listen(
      (position) {
        _currentPosition = position;
        _locationHistory.add(position);
        onLocationUpdate(position);
      },
      onError: (error) {
        onError?.call('位置跟踪错误: $error');
      },
    );
  }
  
  // 停止跟踪
  void stopLocationTracking() {
    _positionSubscription?.cancel();
    _positionSubscription = null;
  }
  
  // 获取移动轨迹
  List<Position> getTrajectory() {
    return List.from(_locationHistory);
  }
  
  // 计算总距离
  double getTotalDistance() {
    if (_locationHistory.length < 2) return 0.0;
    
    double total = 0.0;
    for (int i = 1; i < _locationHistory.length; i++) {
      final prev = _locationHistory[i - 1];
      final current = _locationHistory[i];
      total += LocationService.calculateDistance(
        prev.latitude, prev.longitude,
        current.latitude, current.longitude,
      );
    }
    return total;
  }
  
  // 清除历史记录
  void clearHistory() {
    _locationHistory.clear();
  }
  
  void dispose() {
    stopLocationTracking();
  }
}

    });

      width: 3,

      _polylines.removeWhere((p) => p.polylineId.value == 'trajectory');

    _locationManager.dispose();
    super.dispose();

import 'package:permission_handler/permission_handler.dart';

class PermissionManager {
  // 权限状态枚举
  enum PermissionResult {
    granted,
    denied,
    permanentlyDenied,
    restricted,
  }
  
  // 权限配置
  static const Map<String, Permission> _permissionMap = {
    'camera': Permission.camera,
    'microphone': Permission.microphone,
    'location': Permission.location,
    'locationWhenInUse': Permission.locationWhenInUse,
    'locationAlways': Permission.locationAlways,
    'storage': Permission.storage,
    'photos': Permission.photos,
    'contacts': Permission.contacts,
    'calendar': Permission.calendar,
    'notification': Permission.notification,
  };
  
  // 检查单个权限
  static Future<PermissionResult> checkPermission(String permissionName) async {
    final permission = _permissionMap[permissionName];
    if (permission == null) {
      throw ArgumentError('未知权限: $permissionName');
    }
    
    final status = await permission.status;
    return _convertStatus(status);
  }
  
  // 请求单个权限
  static Future<PermissionResult> requestPermission(String permissionName) async {
    final permission = _permissionMap[permissionName];
    if (permission == null) {
      throw ArgumentError('未知权限: $permissionName');
    }
    
    final status = await permission.request();
    return _convertStatus(status);
  }
  
  // 批量请求权限
  static Future<Map<String, PermissionResult>> requestMultiplePermissions(
    List<String> permissionNames) async {
    
    final permissions = permissionNames
        .map((name) => _permissionMap[name])
        .where((permission) => permission != null)
        .cast<Permission>()
        .toList();
    
    final statuses = await permissions.request();
    
    final results = <String, PermissionResult>{};
    for (final entry in statuses.entries) {
      final permissionName = _permissionMap.entries
          .firstWhere((e) => e.value == entry.key)
          .key;
      results[permissionName] = _convertStatus(entry.value);
    }
    
    return results;
  }
  
  // 检查是否需要显示权限说明
  static Future<bool> shouldShowRequestRationale(String permissionName) async {
    final permission = _permissionMap[permissionName];
    if (permission == null) return false;
    
    return await permission.shouldShowRequestRationale;
  }
  
  // 打开应用设置
  static Future<bool> openAppSettings() async {
    return await openAppSettings();
  }
  
  // 权限状态转换
  static PermissionResult _convertStatus(PermissionStatus status) {
    switch (status) {
      case PermissionStatus.granted:
        return PermissionResult.granted;
      case PermissionStatus.denied:
        return PermissionResult.denied;
      case PermissionStatus.permanentlyDenied:
        return PermissionResult.permanentlyDenied;
      case PermissionStatus.restricted:
        return PermissionResult.restricted;
      default:
        return PermissionResult.denied;
    }
  }
  
  // 权限请求处理器
  static Future<bool> handlePermissionRequest(
    String permissionName, {
    String? rationale,
    VoidCallback? onPermanentlyDenied,
  }) async {
    final currentStatus = await checkPermission(permissionName);
    
    if (currentStatus == PermissionResult.granted) {
      return true;
    }
    
    if (currentStatus == PermissionResult.permanentlyDenied) {
      onPermanentlyDenied?.call();
      return false;
    }
    
    // 显示权限说明（如果需要）
    if (await shouldShowRequestRationale(permissionName) && rationale != null) {
      // 这里可以显示一个对话框解释为什么需要权限
      print('权限说明: $rationale');
    }
    
    final requestResult = await requestPermission(permissionName);
    return requestResult == PermissionResult.granted;
  }
}
