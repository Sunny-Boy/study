# 009. StatelessWidget 与 StatefulWidget

## 学习目标

- 深入理解 StatelessWidget 和 StatefulWidget 的本质区别
- 掌握 State 的生命周期与状态管理机制
- 学会选择合适的 Widget 类型解决实际问题

## 1. StatelessWidget 详解

### 1.1 基本概念

StatelessWidget 是**无状态组件**，其特点：
- 一旦创建，配置信息不可变
- 只能通过父组件传递新的配置来更新
- 性能开销小，适合展示静态内容

### 1.2 StatelessWidget 源码分析

```dart
abstract class StatelessWidget extends Widget {
  const StatelessWidget({super.key});

  @override
  StatelessElement createElement() => StatelessElement(this);

  // 核心方法：构建界面
  Widget build(BuildContext context);
}

// 对应的 Element
class StatelessElement extends ComponentElement {
  StatelessElement(StatelessWidget widget) : super(widget);

  @override
  Widget build() => (widget as StatelessWidget).build(this);

  @override
  void update(StatelessWidget newWidget) {
    super.update(newWidget);
    rebuild();
  }
}
```

## 2. StatefulWidget 详解

### 2.1 基本概念

StatefulWidget 是**有状态组件**，其特点：
- 拥有可变的内部状态
- 状态改变时会触发界面重建
- 通过 State 对象管理生命周期

### 2.2 StatefulWidget 源码分析

```dart
abstract class StatefulWidget extends Widget {
  const StatefulWidget({super.key});

  @override
  StatefulElement createElement() => StatefulElement(this);

  // 核心方法：创建 State 对象
  State createState();
}

// 对应的 Element
class StatefulElement extends ComponentElement {
  StatefulElement(StatefulWidget widget)
      : _state = widget.createState(),
        super(widget) {
    _state._element = this;
    _state._widget = widget;
  }

  State<StatefulWidget> _state;

  @override
  Widget build() => _state.build(this);

  @override
  void update(StatefulWidget newWidget) {
    super.update(newWidget);
    final oldWidget = _state._widget!;
    _state._widget = widget as StatefulWidget;
    _state.didUpdateWidget(oldWidget as StatefulWidget);
    rebuild();
  }
}
```

### 3.2 多计数器管理

```dart
class MultiCounterApp extends StatefulWidget {
  const MultiCounterApp({super.key});

  @override
  State<MultiCounterApp> createState() => _MultiCounterAppState();
}

class _MultiCounterAppState extends State<MultiCounterApp> {
  final List<CounterModel> _counters = [];

  void _addCounter() {
    setState(() {
      _counters.add(CounterModel(
        id: DateTime.now().millisecondsSinceEpoch.toString(),
        name: '计数器 ${_counters.length + 1}',
        value: 0,
      ));
    });
  }

  void _removeCounter(String id) {
    setState(() {
      _counters.removeWhere((counter) => counter.id == id);
    });
  }

  void _updateCounter(String id, int newValue) {
    setState(() {
      final index = _counters.indexWhere((counter) => counter.id == id);
      if (index != -1) {
        _counters[index] = _counters[index].copyWith(value: newValue);
      }
    });
  }

  int get _totalValue => _counters.fold(0, (sum, counter) => sum + counter.value);

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('多计数器管理'),
        actions: [
          Padding(
            padding: const EdgeInsets.all(16),
            child: Center(
              child: Text(
                '总和: $_totalValue',
                style: const TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
              ),
            ),
          ),
        ],
      ),
      body: _counters.isEmpty
          ? const Center(
              child: Text(
                '点击 + 按钮添加计数器',
                style: TextStyle(fontSize: 18, color: Colors.grey),
              ),
            )
          : ListView.builder(
              itemCount: _counters.length,
              itemBuilder: (context, index) {
                final counter = _counters[index];
                return CounterItem(
                  key: ValueKey(counter.id),
                  counter: counter,
                  onValueChanged: (value) => _updateCounter(counter.id, value),
                  onRemove: () => _removeCounter(counter.id),
                );
              },
            ),
      floatingActionButton: FloatingActionButton(
        onPressed: _addCounter,
        child: const Icon(Icons.add),
      ),
    );
  }
}

  final String name;

    required this.id,
    required this.name,
    required this.value,
  });

  CounterModel copyWith({
    String? id,
    String? name,
    int? value,
  }) {
    return CounterModel(
      id: id ?? this.id,
      name: name ?? this.name,
      value: value ?? this.value,
    );
  }
}

  @override
  Widget build(BuildContext context) {
    return Card(

      child: ListTile(
        title: Text(counter.name),
        subtitle: Text('当前值: ${counter.value}'),
        leading: CircleAvatar(
```

## 6. 常见问题与解决方案

### 6.1 setState 使用问题

```dart
class SetStateProblemsDemo extends StatefulWidget {
  const SetStateProblemsDemo({super.key});

  @override
  State<SetStateProblemsDemo> createState() => _SetStateProblemsDemoState();
}

class _SetStateProblemsDemoState extends State<SetStateProblemsDemo> {
  int _counter = 0;
  bool _loading = false;

  // ❌ 错误：在 dispose 后调用 setState
  void _badAsyncOperation() async {
    setState(() => _loading = true);
    
    await Future.delayed(const Duration(seconds: 3));
    
    // 如果用户在等待期间离开页面，这里会报错
    setState(() {
      _loading = false;
      _counter++;
    });
  }

  // ✅ 正确：检查组件是否已销毁
  void _goodAsyncOperation() async {
    if (!mounted) return;
    setState(() => _loading = true);
    
    await Future.delayed(const Duration(seconds: 3));
    
    if (!mounted) return; // 检查组件是否还在
    setState(() {
      _loading = false;
      _counter++;
    });
  }

  // ❌ 错误：在 initState 中直接调用 setState
  @override
  void initState() {
    super.initState();
    // setState(() => _counter = 10); // 这会报错
    
    // ✅ 正确：使用 WidgetsBinding.instance.addPostFrameCallback
    WidgetsBinding.instance.addPostFrameCallback((_) {
      setState(() => _counter = 10);
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('setState 问题演示')),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text('计数: $_counter'),
            if (_loading) const CircularProgressIndicator(),
            const SizedBox(height: 20),
            ElevatedButton(
              onPressed: _loading ? null : _goodAsyncOperation,
              child: const Text('安全的异步操作'),
```

```dart
  State<KeyUsageDemo> createState() => _KeyUsageDemoState();
}

class _KeyUsageDemoState extends State<KeyUsageDemo> {
  List<ItemData> _items = [
    ItemData(id: '1', name: 'Item 1', color: Colors.red),
    ItemData(id: '2', name: 'Item 2', color: Colors.blue),
    ItemData(id: '3', name: 'Item 3', color: Colors.green),
  ];

  void _shuffle() {
    setState(() {
      _items.shuffle();
    });
  }

  void _removeFirst() {
    if (_items.isNotEmpty) {
      setState(() {
        _items.removeAt(0);
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Key 使用演示')),
      body: Column(
        children: [
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceEvenly,
            children: [
              ElevatedButton(onPressed: _shuffle, child: const Text('随机排序')),
              ElevatedButton(onPressed: _removeFirst, child: const Text('删除第一个')),
            ],
          ),
          
          const Divider(),
          
          const Text('❌ 没有使用 Key（状态会错乱）'),
          ..._items.map((item) => StatefulColorBox(item: item)),
          
          const Divider(),
          
          const Text('✅ 正确使用 Key（状态保持正确）'),
          ..._items.map((item) => StatefulColorBox(
            key: ValueKey(item.id),
            item: item,
          )),
        ],
      ),
    );
  }
}

class ItemData {
  final String id;
  final String name;
  final Color color;

  ItemData({required this.id, required this.name, required this.color});
}

class StatefulColorBox extends StatefulWidget {
  const StatefulColorBox({super.key, required this.item});

  final ItemData item;

  @override
  State<StatefulColorBox> createState() => _StatefulColorBoxState();
}

class _StatefulColorBoxState extends State<StatefulColorBox> {
  late Color _currentColor;

  @override
  void initState() {
    super.initState();
    _currentColor = widget.item.color;
  }

  void _changeColor() {
    setState(() {
      _currentColor = _currentColor == widget.item.color 
          ? Colors.grey 
          : widget.item.color;
    });
  }

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTap: _changeColor,
      child: Container(
        margin: const EdgeInsets.all(4),
        padding: const EdgeInsets.all(16),
        color: _currentColor,
        child: Text(
          widget.item.name,
          style: const TextStyle(color: Colors.white),
        ),
      ),
    );
  }
}
```

## 7. 总结

### 7.1 选择指南

| 使用场景 | 推荐类型 | 原因 |
|----------|----------|------|
| 纯展示内容 | StatelessWidget | 性能好，简单 |
| 需要交互 | StatefulWidget | 可管理状态 |
| 频繁重建 | StatelessWidget + 外部状态 | 避免不必要重建 |
| 复杂状态 | 状态管理库 | 更好的可维护性 |

### 7.2 核心要点

1. **StatelessWidget**: 无状态，配置不变，性能优秀
2. **StatefulWidget**: 有状态，可交互，管理生命周期
3. **State 生命周期**: initState → build → didUpdateWidget → dispose
4. **setState**: 触发重建，需注意异步调用和组件销毁
5. **Key**: 帮助 Framework 正确识别和复用 Element

### 7.3 最佳实践

- 优先使用 StatelessWidget
- 合理提升状态到合适的层级
- 使用 const 构造函数优化性能
- 正确处理异步操作中的 setState
- 在需要时使用合适的 Key
