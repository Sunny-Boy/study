# 011. 布局系统与约束机制

## 学习目标

- 深入理解 Flutter 布局系统的约束传递机制
- 掌握核心布局组件的使用方法与原理

## 1. Flutter 布局原理

### 1.1 约束传递系统

Flutter 布局系统遵循一个核心规则：**约束向下传递，尺寸向上返回**

```dart
class ConstraintDemo extends StatelessWidget {
  const ConstraintDemo({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('约束传递演示')),
      body: Column(
        children: [
          // 1. 无约束容器
          Container(
            color: Colors.red.shade100,
            child: const Text('我会尽可能小'),
          ),
          
          const SizedBox(height: 16),
          
          // 2. 有宽度约束
          SizedBox(
            width: 200,
            child: Container(
              color: Colors.blue.shade100,
              child: const Text('我被限制了宽度'),
            ),
          ),
          
          const SizedBox(height: 16),
          
          // 3. 充满父容器
          Expanded(
            child: Container(
              color: Colors.green.shade100,
              child: const Center(
                child: Text('我会填充剩余空间'),
              ),
            ),
          ),
        ],
      ),
    );
  }
}
```

```dart
                children: [
                  Row(
                    children: [

                        value: _direction,
                        onChanged: (value) => setState(() => _direction = value!),
                        items: const [
                          DropdownMenuItem(value: Axis.horizontal, child: Text('水平')),
                          DropdownMenuItem(value: Axis.vertical, child: Text('垂直')),
                        ],
                      ),
                    ],
                  ),
                  
                  Row(

                            DropdownMenuItem(
                              value: alignment,
                              child: Text(alignment.toString().split('.').last),
                            ),
                          ).toList(),
                        ),
                      ),
                    ],
                  ),
                  
                  Row(
                    children: [
                      const Text('交叉轴对齐: '),
                      Expanded(
                        child: DropdownButton<CrossAxisAlignment>(

                            alignment != CrossAxisAlignment.baseline

                            DropdownMenuItem(
```

```dart
class RenderCenterScale extends RenderShiftedBox {
  RenderCenterScale({
    required double scale,
    RenderBox? child,
  }) : _scale = scale,
       super(child);

  double _scale;
  
  double get scale => _scale;
  set scale(double value) {
    if (_scale != value) {
      _scale = value;
      markNeedsLayout();
    }
  }

  @override
  void performLayout() {
    if (child != null) {
      // 让子组件使用原始约束进行布局
      child!.layout(constraints, parentUsesSize: true);
      
      // 计算缩放后的尺寸
      final scaledWidth = child!.size.width * scale;
      final scaledHeight = child!.size.height * scale;
      
      // 设置自己的大小为缩放后的大小
      size = Size(scaledWidth, scaledHeight);
    } else {
      size = constraints.smallest;
    }
  }

  @override
  void paint(PaintingContext context, Offset offset) {
    if (child != null) {
      // 计算居中偏移
      final centerOffset = Offset(
        (size.width - child!.size.width * scale) / 2,
        (size.height - child!.size.height * scale) / 2,
      );
      
      // 应用缩放变换
      context.pushTransform(
        needsCompositing,
        offset + centerOffset,
        Matrix4.diagonal3Values(scale, scale, 1.0),
        (context, offset) {
          context.paintChild(child!, offset);
        },
      );
    }
  }

  @override
  bool hitTestChildren(BoxHitTestResult result, {required Offset position}) {
    if (child != null) {
      // 反向变换坐标用于命中测试
      final centerOffset = Offset(
        (size.width - child!.size.width * scale) / 2,
        (size.height - child!.size.height * scale) / 2,
      );
      
      final transform = Matrix4.diagonal3Values(1.0 / scale, 1.0 / scale, 1.0);
      final childPosition = MatrixUtils.transformPoint(
        transform,
        position - centerOffset,
      );
      
      return child!.hitTest(result, position: childPosition);
    }
    return false;
  }
}
```

```dart
class RenderFlowTagLayout extends RenderBox
    with ContainerRenderObjectMixin<RenderBox, FlowTagLayoutParentData>,
         RenderBoxContainerDefaultsMixin<RenderBox, FlowTagLayoutParentData> {
  
  RenderFlowTagLayout({
    required double spacing,
    required double runSpacing,
  }) : _spacing = spacing,
       _runSpacing = runSpacing;

  double _spacing;
  double _runSpacing;

  double get spacing => _spacing;
  set spacing(double value) {
    if (_spacing != value) {
      _spacing = value;
      markNeedsLayout();
    }
  }

  double get runSpacing => _runSpacing;
  set runSpacing(double value) {
    if (_runSpacing != value) {
      _runSpacing = value;
      markNeedsLayout();
    }
  }

  @override
  void setupParentData(RenderBox child) {
    if (child.parentData is! FlowTagLayoutParentData) {
      child.parentData = FlowTagLayoutParentData();
    }
  }

  @override
  void performLayout() {
    if (firstChild == null) {
      size = constraints.smallest;
      return;
    }

    double width = 0;
    double height = 0;
    double currentLineWidth = 0;
    double currentLineHeight = 0;
    double maxWidth = constraints.maxWidth;

    // 第一次遍历：计算每个子组件的大小
    RenderBox? child = firstChild;
    while (child != null) {
      child.layout(constraints.loosen(), parentUsesSize: true);
      child = childAfter(child);
    }

    // 第二次遍历：安排位置
    child = firstChild;
    double x = 0;
    double y = 0;
    currentLineHeight = 0;

    while (child != null) {
      final childParentData = child.parentData! as FlowTagLayoutParentData;
      final childWidth = child.size.width;
      final childHeight = child.size.height;

      // 检查是否需要换行
      if (x + childWidth > maxWidth && x > 0) {
        // 换行
        x = 0;
        y += currentLineHeight + runSpacing;
        currentLineHeight = 0;
      }

      // 设置子组件位置
      childParentData.offset = Offset(x, y);
      
      // 更新当前行的宽度和高度
      x += childWidth + spacing;
      currentLineWidth = math.max(currentLineWidth, x - spacing);
      currentLineHeight = math.max(currentLineHeight, childHeight);
      
      child = childAfter(child);
    }

    // 设置自己的大小
    width = currentLineWidth;
    height = y + currentLineHeight;
    size = Size(width, height);
  }

  @override
  void paint(PaintingContext context, Offset offset) {
    defaultPaint(context, offset);
  }

  @override
  bool hitTestChildren(BoxHitTestResult result, {required Offset position}) {
    return defaultHitTestChildren(result, position: position);
  }
}

class FlowTagLayoutParentData extends ContainerBoxParentData<RenderBox> {}
```

## 8. 核心要点总结

### 8.1 布局原则

1. **约束传递**: 约束向下，尺寸向上
2. **布局选择**: 根据需求选择合适的布局组件
3. **性能优化**: 减少嵌套，合理使用 const
4. **响应式**: 使用 LayoutBuilder 适配不同屏幕

### 8.2 常用布局组件

| 组件 | 用途 | 特点 |
|------|------|------|
| Row/Column | 线性布局 | 主轴和交叉轴对齐 |
| Stack | 层叠布局 | 绝对定位或相对定位 |
| Flex | 弹性布局 | 动态方向的线性布局 |
| Wrap | 自动换行 | 超出边界自动换行 |
| LayoutBuilder | 响应式布局 | 根据约束动态构建 |

### 8.3 最佳实践

- 理解约束传递机制
- 选择最简单的布局方案
- 注意性能优化
- 考虑多设备适配
- 合理使用自定义布局
