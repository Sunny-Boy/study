# 015 - Bloc/Cubit 状态管理

## 概述

Bloc (Business Logic Component) 是一种强大的状态管理方案，基于 Stream 和 Reactive Programming 构建。Cubit 是 Bloc 的简化版本。本章将深入学习 Bloc/Cubit 的设计思想、核心概念以及在实际项目中的应用。

## Bloc 设计思想与原理

### 核心概念

Bloc 采用单向数据流和事件驱动的架构模式：

- **Event** - 输入事件，描述发生了什么
- **State** - 状态，描述当前的应用状态  
- **Bloc** - 业务逻辑组件，接收事件并产生新状态
- **Transition** - 状态转换，从一个状态到另一个状态的过程

class CounterState extends Equatable {

  final bool isLoading;

    this.count = 0,
    this.isLoading = false,
  });

  List<Object> get props => [count, isLoading];

// 3. 定义 Bloc
class CounterBloc extends Bloc<CounterEvent, CounterState> {
  CounterBloc() : super(CounterState()) {
    // 注册事件处理器
    on<CounterIncremented>(_onIncremented);
    on<CounterDecremented>(_onDecremented);
    on<CounterReset>(_onReset);
    on<CounterSetValue>(_onSetValue);

  void _onIncremented(
    CounterIncremented event,
    Emitter<CounterState> emit,
  ) {
    emit(state.copyWith(count: state.count + 1));
  }
  
  void _onDecremented(
    CounterDecremented event,
    Emitter<CounterState> emit,

    emit(state.copyWith(count: state.count - 1));

  void _onReset(

  }
  
  void _onSetValue(
    CounterSetValue event,
    Emitter<CounterState> emit,
  ) {
    emit(state.copyWith(count: event.value));
  }
}

## Cubit 简化版本

### Cubit 基础用法

Cubit 是 Bloc 的简化版本，不需要事件，直接调用方法改变状态：

// 使用 Cubit 的计数器
class CounterCubit extends Cubit<int> {
  CounterCubit() : super(0);
  
  void increment() => emit(state + 1);
  void decrement() => emit(state - 1);
  void reset() => emit(0);
  
  void setValue(int value) => emit(value);
}

// 复杂状态的 Cubit
class TimerState extends Equatable {
  final int duration;
  final bool isRunning;
  final bool isCompleted;
  
  const TimerState({
    required this.duration,
    this.isRunning = false,
    this.isCompleted = false,
  });
  
  TimerState copyWith({
    int? duration,
    bool? isRunning,
    bool? isCompleted,
  }) {
    return TimerState(
      duration: duration ?? this.duration,
      isRunning: isRunning ?? this.isRunning,
      isCompleted: isCompleted ?? this.isCompleted,
    );
  }
  
  @override
  List<Object> get props => [duration, isRunning, isCompleted];
}

class TimerCubit extends Cubit<TimerState> {
  Timer? _timer;
  
  TimerCubit() : super(TimerState(duration: 60));
  
  void startTimer() {
    if (state.isRunning) return;
    
    emit(state.copyWith(isRunning: true, isCompleted: false));
    
    _timer = Timer.periodic(Duration(seconds: 1), (timer) {
      if (state.duration > 0) {
        emit(state.copyWith(duration: state.duration - 1));
      } else {
        _timer?.cancel();
        emit(state.copyWith(
          isRunning: false,
          isCompleted: true,
        ));
      }
    });
  }
  
  void pauseTimer() {
    _timer?.cancel();
    emit(state.copyWith(isRunning: false));
  }
  
  void resetTimer({int? newDuration}) {
    _timer?.cancel();
    emit(TimerState(duration: newDuration ?? 60));
  }
  
  @override
  Future<void> close() {
    _timer?.cancel();
    return super.close();
  }
}

// 认证 Bloc
class AuthBloc extends Bloc<AuthEvent, AuthState> {
  final AuthRepository _authRepository;
  late StreamSubscription<AuthStatus> _authStatusSubscription;
  
  AuthBloc({required AuthRepository authRepository})
      : _authRepository = authRepository,
        super(const AuthState()) {
    
    // 监听认证状态变化
    _authStatusSubscription = _authRepository.status.listen(
      (status) => add(AuthStatusChanged(status)),
    );
    
    on<AuthLoginRequested>(_onLoginRequested);
    on<AuthLogoutRequested>(_onLogoutRequested);
    on<AuthStatusChanged>(_onAuthStatusChanged);
    on<AuthTokenRefreshRequested>(_onTokenRefreshRequested);
  }
  
  Future<void> _onLoginRequested(
    AuthLoginRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(state.copyWith(isLoading: true, errorMessage: null));
    
    try {
      final user = await _authRepository.login(
        username: event.username,
        password: event.password,
      );
      
      emit(state.copyWith(
        status: AuthStatus.authenticated,
        user: user,
        isLoading: false,
      ));
    } catch (e) {
      emit(state.copyWith(
        status: AuthStatus.unauthenticated,
        errorMessage: e.toString(),
        isLoading: false,
      ));
    }
  }
  
  void _onLogoutRequested(
    AuthLogoutRequested event,
    Emitter<AuthState> emit,
  ) {
    _authRepository.logout();
  }
  
  void _onAuthStatusChanged(
    AuthStatusChanged event,
    Emitter<AuthState> emit,
  ) {
    switch (event.status) {
      case AuthStatus.authenticated:
        final user = _authRepository.currentUser;
        emit(state.copyWith(
          status: AuthStatus.authenticated,
          user: user,
          isLoading: false,
        ));
        break;
      case AuthStatus.unauthenticated:
        emit(state.copyWith(
          status: AuthStatus.unauthenticated,
          user: null,
          isLoading: false,
        ));
        break;
      case AuthStatus.unknown:
        emit(state.copyWith(
          status: AuthStatus.unknown,
          isLoading: true,
        ));
        break;
    }
  }
  
  Future<void> _onTokenRefreshRequested(
    AuthTokenRefreshRequested event,
    Emitter<AuthState> emit,
  ) async {
    try {
      await _authRepository.refreshToken();
    } catch (e) {
      add(AuthLogoutRequested());
    }
  }
  
  @override
  Future<void> close() {
    _authStatusSubscription.cancel();
    return super.close();
  }
}

// 搜索 Bloc 使用 StreamTransformer
class SearchEvent extends Equatable {
  final String query;
  
  const SearchEvent(this.query);
  
  @override
  List<Object> get props => [query];
}

class SearchState extends Equatable {
  final List<String> results;
  final bool isLoading;
  final String? errorMessage;
  
  const SearchState({
    this.results = const [],
    this.isLoading = false,
    this.errorMessage,
  });
  
  SearchState copyWith({
    List<String>? results,
    bool? isLoading,
    String? errorMessage,
  }) {
    return SearchState(
      results: results ?? this.results,
      isLoading: isLoading ?? this.isLoading,
      errorMessage: errorMessage,
    );
  }
  
  @override
  List<Object?> get props => [results, isLoading, errorMessage];
}

class SearchBloc extends Bloc<SearchEvent, SearchState> {
  SearchBloc() : super(const SearchState()) {
    // 使用 debounce 防抖
    on<SearchEvent>(
      _onSearchEvent,
      transformer: debounce(Duration(milliseconds: 500)),
    );
  }
  
  Future<void> _onSearchEvent(
    SearchEvent event,
    Emitter<SearchState> emit,
  ) async {
    if (event.query.isEmpty) {
      emit(state.copyWith(results: [], isLoading: false));
      return;
    }
    
    emit(state.copyWith(isLoading: true, errorMessage: null));
    
    try {
      final results = await _performSearch(event.query);
      emit(state.copyWith(results: results, isLoading: false));
    } catch (e) {
      emit(state.copyWith(
        errorMessage: e.toString(),
        isLoading: false,
        results: [],
      ));
    }
  }
  
  Future<List<String>> _performSearch(String query) async {
    await Future.delayed(Duration(milliseconds: 800));
    
    // 模拟搜索结果
    final allItems = [
      'Apple', 'Banana', 'Cherry', 'Date', 'Elderberry',
      'Fig', 'Grape', 'Honeydew', 'Kiwi', 'Lemon'
    ];
    
    return allItems
        .where((item) => item.toLowerCase().contains(query.toLowerCase()))
        .toList();
  }
}

// 自定义 debounce 转换器
EventTransformer<E> debounce<E>(Duration duration) {
  return (events, mapper) {
    return events.debounceTime(duration).switchMap(mapper);
  };
}

// 仓储实现
class TodoRepositoryImpl implements TodoRepository {
  final List<TodoModel> _todos = [];
  
  @override
  Future<List<Todo>> getTodos() async {
    await Future.delayed(Duration(milliseconds: 500)); // 模拟延迟
    return _todos.map((model) => model.toDomain()).toList();
  }
  
  @override
  Future<Todo> getTodoById(String id) async {
    await Future.delayed(Duration(milliseconds: 200));
    final model = _todos.firstWhere((todo) => todo.id == id);
    return model.toDomain();
  }
  
  @override
  Future<void> addTodo(Todo todo) async {
    await Future.delayed(Duration(milliseconds: 300));
    _todos.add(TodoModel.fromDomain(todo));
  }
  
  @override
  Future<void> updateTodo(Todo todo) async {
    await Future.delayed(Duration(milliseconds: 300));
    final index = _todos.indexWhere((t) => t.id == todo.id);
    if (index != -1) {
      _todos[index] = TodoModel.fromDomain(todo);
    }
  }
  
  @override
  Future<void> deleteTodo(String id) async {
    await Future.delayed(Duration(milliseconds: 200));
    _todos.removeWhere((todo) => todo.id == id);
  }
}

  final String title;
  final String description;
  
  TodoAdded({required this.title, required this.description});
  
  @override
  List<Object> get props => [title, description];
}

class TodoCompleted extends TodoBlocEvent {
  final String id;
  
  TodoCompleted(this.id);
  
  @override
  List<Object> get props => [id];
}

class TodoDeleted extends TodoBlocEvent {
  final String id;
  
  TodoDeleted(this.id);
  
  @override
  List<Object> get props => [id];
}

// Bloc 状态
class TodoBlocState extends Equatable {
  final List<Todo> todos;
  final bool isLoading;
  final String? errorMessage;
  
  const TodoBlocState({
    this.todos = const [],
    this.isLoading = false,
    this.errorMessage,
  });
  
  TodoBlocState copyWith({
    List<Todo>? todos,
    bool? isLoading,
    String? errorMessage,
  }) {
    return TodoBlocState(
      todos: todos ?? this.todos,
      isLoading: isLoading ?? this.isLoading,
      errorMessage: errorMessage,
    );
  }
  
  @override
  List<Object?> get props => [todos, isLoading, errorMessage];
}

// Todo Bloc
class TodoBloc extends Bloc<TodoBlocEvent, TodoBlocState> {
  final GetTodosUseCase getTodosUseCase;
  final AddTodoUseCase addTodoUseCase;
  final CompleteTodoUseCase completeTodoUseCase;
  final TodoRepository todoRepository;
  
  TodoBloc({
    required this.getTodosUseCase,
    required this.addTodoUseCase,
    required this.completeTodoUseCase,
    required this.todoRepository,
  }) : super(const TodoBlocState()) {
    on<TodosLoadRequested>(_onTodosLoadRequested);
    on<TodoAdded>(_onTodoAdded);
    on<TodoCompleted>(_onTodoCompleted);
    on<TodoDeleted>(_onTodoDeleted);
  }
  
  Future<void> _onTodosLoadRequested(
    TodosLoadRequested event,
    Emitter<TodoBlocState> emit,
  ) async {
    emit(state.copyWith(isLoading: true));
    
    try {
      final todos = await getTodosUseCase.execute();
      emit(state.copyWith(todos: todos, isLoading: false));
    } catch (e) {
      emit(state.copyWith(
        errorMessage: e.toString(),
        isLoading: false,
      ));
    }
  }
  
  Future<void> _onTodoAdded(
    TodoAdded event,
    Emitter<TodoBlocState> emit,
  ) async {
    try {
      await addTodoUseCase.execute(event.title, event.description);
      add(TodosLoadRequested());
    } catch (e) {
      emit(state.copyWith(errorMessage: e.toString()));
    }
  }
  
  Future<void> _onTodoCompleted(
    TodoCompleted event,
    Emitter<TodoBlocState> emit,
  ) async {
    try {
      await completeTodoUseCase.execute(event.id);
      add(TodosLoadRequested());
    } catch (e) {
      emit(state.copyWith(errorMessage: e.toString()));
    }
  }
  
  Future<void> _onTodoDeleted(
    TodoDeleted event,
    Emitter<TodoBlocState> emit,
  ) async {
    try {
      await todoRepository.deleteTodo(event.id);
      add(TodosLoadRequested());
    } catch (e) {
      emit(state.copyWith(errorMessage: e.toString()));
    }
  }
}

## 最佳实践总结

### 1. 何时选择 Bloc vs Cubit

- **使用 Cubit**：
  - 简单的状态管理需求
  - 状态变化逻辑简单
  - 团队对响应式编程不熟悉

- **使用 Bloc**：
  - 复杂的业务逻辑
  - 需要事件溯源和调试
  - 大型项目和团队开发

### 2. 状态设计原则

```dart
// ✅ 好的状态设计
class UserState extends Equatable {
  final User? user;
  final bool isLoading;
  final String? errorMessage;
  
  // 使用 copyWith 方法
  // 实现 Equatable 接口
  // 状态不可变
}

// ❌ 避免的设计
class BadState {
  User? user; // 可变状态
  bool isLoading = false;
  // 没有实现 Equatable
  // 没有 copyWith 方法
}
```

### 3. 事件设计原则

```dart
// ✅ 具体明确的事件
abstract class UserEvent extends Equatable {}

class UserLoginRequested extends UserEvent {
  final String email;
  final String password;
  // 包含必要的数据
}

class UserLogoutRequested extends UserEvent {}

// ❌ 过于宽泛的事件
class UserAction extends UserEvent {
  final String action; // 不明确
  final Map<String, dynamic> data; // 类型不安全
}
```

## 总结

Bloc/Cubit 是功能强大的状态管理方案：

1. **清晰的架构** - 单向数据流，职责分离
2. **可测试性** - 易于编写单元测试
3. **可扩展性** - 适合大型项目
4. **调试友好** - 完整的状态变化日志
5. **社区支持** - 丰富的生态系统

选择 Bloc/Cubit 时要考虑项目复杂度和团队技术栈，对于简单应用可能过度设计，但对于复杂的企业级应用是很好的选择。
