# 013 - 状态提升与数据传递

## 概述

当多个组件需要共享状态时，我们需要将状态"提升"到它们的共同父组件中。本章将深入学习状态提升设计模式、InheritedWidget 机制以及 Context 的数据传递原理。

## 状态提升设计模式

### 基本概念

状态提升是指将子组件的状态移动到父组件中管理，然后通过回调函数和属性传递来实现数据共享。

### 简单状态提升示例

```dart
// 父组件管理共享状态
class ShoppingCart extends StatefulWidget {
  @override
  _ShoppingCartState createState() => _ShoppingCartState();
}

class _ShoppingCartState extends State<ShoppingCart> {
  final List<Product> _cartItems = [];
  double _totalPrice = 0.0;

  void _addToCart(Product product) {
    setState(() {
      _cartItems.add(product);
      _totalPrice += product.price;
    });
  }

  void _removeFromCart(Product product) {
    setState(() {
      _cartItems.remove(product);
      _totalPrice -= product.price;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('购物车')),
      body: Column(
        children: [
          // 产品列表组件
          ProductList(
            onAddToCart: _addToCart,
          ),
          // 购物车组件
          CartSummary(
            items: _cartItems,
            totalPrice: _totalPrice,
            onRemoveFromCart: _removeFromCart,
          ),
        ],
      ),
    );
  }
}
```

### 复杂状态提升示例

```dart
class TodoApp extends StatefulWidget {
  @override
  _TodoAppState createState() => _TodoAppState();
}

class _TodoAppState extends State<TodoApp> {
  List<Todo> _todos = [];
  String _filter = 'all'; // all, active, completed

  void _addTodo(String title) {
    setState(() {
      _todos.add(Todo(
        id: DateTime.now().millisecondsSinceEpoch,
        title: title,
        completed: false,
      ));
    });
  }

  void _toggleTodo(int id) {
    setState(() {
      final index = _todos.indexWhere((todo) => todo.id == id);
      if (index != -1) {
        _todos[index] = _todos[index].copyWith(
          completed: !_todos[index].completed,
        );
      }
    });
  }

  void _deleteTodo(int id) {
    setState(() {
      _todos.removeWhere((todo) => todo.id == id);
    });
  }

  void _setFilter(String filter) {
    setState(() {
      _filter = filter;
    });
  }

  List<Todo> get _filteredTodos {
    switch (_filter) {
      case 'active':
        return _todos.where((todo) => !todo.completed).toList();
      case 'completed':
        return _todos.where((todo) => todo.completed).toList();
      default:
        return _todos;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('待办事项')),
      body: Column(
        children: [
          // 添加待办事项
          TodoInput(onAddTodo: _addTodo),
          // 过滤器
          TodoFilter(
            currentFilter: _filter,
            onFilterChanged: _setFilter,
          ),
          // 待办事项列表
          Expanded(
            child: TodoList(
              todos: _filteredTodos,
              onToggleTodo: _toggleTodo,
              onDeleteTodo: _deleteTodo,
            ),
          ),
          // 统计信息
          TodoStats(
            totalCount: _todos.length,
            activeCount: _todos.where((todo) => !todo.completed).length,
            completedCount: _todos.where((todo) => todo.completed).length,
          ),
        ],
      ),
    );
  }
}

class Todo {
  final int id;
  final String title;
  final bool completed;

  Todo({required this.id, required this.title, required this.completed});

  Todo copyWith({String? title, bool? completed}) {
    return Todo(
      id: id,
      title: title ?? this.title,
      completed: completed ?? this.completed,
    );
  }
}

class TodoInput extends StatefulWidget {
  final Function(String) onAddTodo;

  TodoInput({required this.onAddTodo});

  @override
  _TodoInputState createState() => _TodoInputState();
}

class _TodoInputState extends State<TodoInput> {
  final _controller = TextEditingController();

  void _submitTodo() {
    final title = _controller.text.trim();
    if (title.isNotEmpty) {
      widget.onAddTodo(title);
      _controller.clear();
    }
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: EdgeInsets.all(16),
      child: Row(
        children: [
          Expanded(
            child: TextField(
              controller: _controller,
              decoration: InputDecoration(
                hintText: '输入待办事项...',
                border: OutlineInputBorder(),
              ),
              onSubmitted: (_) => _submitTodo(),
            ),
          ),
          SizedBox(width: 8),
          ElevatedButton(
            onPressed: _submitTodo,
            child: Text('添加'),
          ),
        ],
      ),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }
}

class TodoFilter extends StatelessWidget {

          value: 'completed',

    );

class TodoList extends StatelessWidget {

  final Function(int) onToggleTodo;

  });

                : null,
            ),

          trailing: IconButton(
            icon: Icon(Icons.delete, color: Colors.red),
            onPressed: () => onDeleteTodo(todo.id),
          ),
        );
      },
    );
  }
}
```

## InheritedWidget 深入解析

### InheritedWidget 基础

InheritedWidget 是一种特殊的 Widget，可以高效地向下传递数据给子组件树中的任意组件。

```dart
class UserData extends InheritedWidget {
  final User user;
  final Function(User) updateUser;

  UserData({
    Key? key,
    required this.user,
    required this.updateUser,
    required Widget child,
  }) : super(key: key, child: child);

  static UserData? of(BuildContext context) {
    return context.dependOnInheritedWidgetOfExactType<UserData>();
  }

  @override
  bool updateShouldNotify(UserData oldWidget) {
    return user != oldWidget.user;
  }
}

class User {
  final String name;
  final String email;
  final int age;

  User({required this.name, required this.email, required this.age});

  User copyWith({String? name, String? email, int? age}) {
    return User(
      name: name ?? this.name,
      email: email ?? this.email,
      age: age ?? this.age,
    );
  }

  @override
  bool operator ==(Object other) {
    if (identical(this, other)) return true;
    return other is User &&
        other.name == name &&
        other.email == email &&
        other.age == age;
  }

  @override
  int get hashCode => name.hashCode ^ email.hashCode ^ age.hashCode;
}
```

### InheritedModel 进阶使用

InheritedModel 是 InheritedWidget 的扩展，支持细粒度的更新控制。

```dart
enum UserAspect { profile, settings, preferences }

class UserModel extends InheritedModel<UserAspect> {
  final User user;
  final UserPreferences preferences;
  final Function(User) updateUser;
  final Function(UserPreferences) updatePreferences;

  UserModel({
    Key? key,
    required this.user,
    required this.preferences,
    required this.updateUser,
    required this.updatePreferences,
    required Widget child,
  }) : super(key: key, child: child);

  static UserModel? of(BuildContext context, {UserAspect? aspect}) {
    return InheritedModel.inheritFrom<UserModel>(context, aspect: aspect);
  }

  @override
  bool updateShouldNotify(UserModel oldWidget) {
    return user != oldWidget.user || preferences != oldWidget.preferences;
  }

  @override
  bool updateShouldNotifyDependent(UserModel oldWidget, Set<UserAspect> dependencies) {
    return dependencies.any((aspect) {
      switch (aspect) {
        case UserAspect.profile:
          return user != oldWidget.user;
        case UserAspect.settings:
        case UserAspect.preferences:
          return preferences != oldWidget.preferences;
      }
    });
  }
}

class UserPreferences {
  final bool darkMode;
  final String language;
  final bool notifications;

  UserPreferences({
    required this.darkMode,
    required this.language,
    required this.notifications,
  });

  UserPreferences copyWith({bool? darkMode, String? language, bool? notifications}) {
    return UserPreferences(
      darkMode: darkMode ?? this.darkMode,
      language: language ?? this.language,
      notifications: notifications ?? this.notifications,
    );
  }

  @override
  bool operator ==(Object other) {
    if (identical(this, other)) return true;
    return other is UserPreferences &&
        other.darkMode == darkMode &&
        other.language == language &&
        other.notifications == notifications;
  }

  @override
  int get hashCode => darkMode.hashCode ^ language.hashCode ^ notifications.hashCode;
}
```

### 自定义 Context 扩展

```dart
extension ContextExtensions on BuildContext {
  // 便捷的主题访问
  ThemeData get theme => Theme.of(this);
  TextTheme get textTheme => theme.textTheme;
  ColorScheme get colorScheme => theme.colorScheme;
  
  // 便捷的媒体查询访问
  MediaQueryData get mediaQuery => MediaQuery.of(this);
  Size get screenSize => mediaQuery.size;
  EdgeInsets get padding => mediaQuery.padding;
  EdgeInsets get viewInsets => mediaQuery.viewInsets;
  
  // 便捷的导航访问
```

```dart
class ThemeProvider extends InheritedWidget {
  final ThemeData themeData;
  final bool isDarkMode;
  final Function() toggleTheme;

  ThemeProvider({
    Key? key,
    required this.themeData,
    required this.isDarkMode,
    required this.toggleTheme,
    required Widget child,
  }) : super(key: key, child: child);

  static ThemeProvider? of(BuildContext context) {
    return context.dependOnInheritedWidgetOfExactType<ThemeProvider>();
  }

  @override
  bool updateShouldNotify(ThemeProvider oldWidget) {
    return themeData != oldWidget.themeData;
  }
}
```

## 最佳实践总结

### 1. 选择合适的状态管理方案

- **局部状态** - 使用 setState
- **跨几个组件** - 使用状态提升
- **跨多个页面** - 使用 InheritedWidget
- **复杂应用** - 使用专门的状态管理库（Provider、Bloc 等）

### 2. 性能优化建议

```dart
// ✅ 好的做法：细粒度控制
class OptimizedWidget extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        // 只监听特定数据的组件
        Consumer<SpecificData>(
          builder: (context, data, child) {
            return Text(data.value);
          },
        ),
        // 静态组件使用 const
        const StaticComponent(),
      ],
    );
  }
}

// ❌ 避免：过度监听
class BadWidget extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    // 监听整个应用状态，导致不必要的重建
    final appState = AppState.of(context);
    
    return Text(appState.someUnrelatedData);
  }
}
```

### 3. 数据流向设计

```dart
// 单向数据流
Parent Widget (State)
    ↓ (Props)
Child Widget
    ↓ (Events)
Callbacks
    ↓ (State Changes)
Parent Widget (Update)
```

## 总结

状态提升和数据传递是 Flutter 应用架构的基础：

1. **状态提升** - 将共享状态移到公共父组件
2. **InheritedWidget** - 高效的向下数据传递机制
3. **Context** - 访问祖先组件和框架服务的桥梁
4. **性能优化** - 最小化重建范围，合理使用 const
5. **架构设计** - 保持单向数据流，职责分离
